    /**
     * Re-entrant per-zone platformer-physics override (D-12). Fields set in this
     * block SHADOW the game-level platformerPhysics config for the active zone.
     * Fields NOT set in this block fall through to the game-level defaults.
     *
     * Example:
     * ```
     * zone("world2Area1Zone") {
     *     tileset(asset("res/graphics/world2-tileset.png"))
     *     tiles(asset("res/graphics/world2-area1.png"))
     *     platformerPhysics {
     *         gravity(3)        // heavier in world 2
     *         solidThreshold(68)
     *     }
     * }
     * ```
     */
    fun ZoneBuilder.platformerPhysics(block: PlatformerPhysicsBuilder.() -> Unit) {
        // Use a marker builder that exposes which fields were set.
        val builder = OverrideTrackingPhysicsBuilder()
        builder.block()
        setPlatformerPhysicsOverride(builder.toOverrideMap())
    }
    ```

    Then add the helper class at module-internal scope (same file, package-private):
    ```kotlin
    /**
     * PlatformerPhysicsBuilder variant that tracks WHICH setters were invoked,
     * so per-level overrides can shadow only the explicitly-set fields.
     * Keys in the resulting map use the PlatformerPhysicsConfig field names verbatim.
     */
    internal class OverrideTrackingPhysicsBuilder : PlatformerPhysicsBuilder() {
        private val setFields = mutableMapOf<String, Any>()

        override fun gravity(value: Int) { super.gravity(value); setFields["gravity"] = value }
        override fun jumpForce(value: Int) { super.jumpForce(value); setFields["jumpForce"] = value }
        override fun terminalVelocity(value: Int) { super.terminalVelocity(value); setFields["terminalVelocity"] = value }
        override fun solidThreshold(value: Int) { super.solidThreshold(value); setFields["solidThreshold"] = value }
        override fun jumpHold(maxFrames: Int) { super.jumpHold(maxFrames); setFields["jumpHoldMaxFrames"] = maxFrames }
        // Add overrides for any other PlatformerPhysicsBuilder method the planner / user
        // is likely to use for per-level shadowing. If a method is NOT overridden here,
        // its set-call falls through to super and is NOT recorded in the override map
        // (it becomes a game-level-only setting — that's intentional for fields like
        // wallJump / variableHeightJump which are not currently per-level overridable).

        fun toOverrideMap(): Map<String, Any> = setFields.toMap()
    }
    ```

    **IMPORTANT — open/abstract changes required:** the existing `PlatformerPhysicsBuilder` methods in `PlatformerBuilders.kt` (gravity, jumpForce, terminalVelocity, solidThreshold, jumpHold) MUST be made `open` so OverrideTrackingPhysicsBuilder can override them. If they are currently not `open`, add the `open` modifier to each in this same plan (PlatformerBuilders.kt modification — same wave-1 plan, no new file). This is a single-keyword change per method, ~5 keywords total.

    Note: The class is `internal` to keep the extension-builder helper invisible to end-users (they call `platformerPhysics { }`, not `OverrideTrackingPhysicsBuilder` directly).
