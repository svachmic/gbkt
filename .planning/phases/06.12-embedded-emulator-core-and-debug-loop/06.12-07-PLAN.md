Create `LogCatPanel` as a JPanel that can be hosted in a JFrame:

```kotlin
class LogCatPanel : JPanel(BorderLayout()) {
    private val textArea: JTextArea
    private val scrollPane: JScrollPane
    private val filterCombo: JComboBox<String>
    private val clearButton: JButton
    private val entries = mutableListOf<DebugLogEntry>()
    private var currentFilter: LogLevel? = null  // null = ALL

    init {
        // Text area - terminal style
        textArea = JTextArea().apply {
            font = Font(Font.MONOSPACED, Font.PLAIN, 12)
            background = Color(30, 30, 30)
            foreground = Color(220, 220, 220)
            caretColor = Color(220, 220, 220)
            isEditable = false
            lineWrap = false  // No wrapping - horizontal scroll
        }
        scrollPane = JScrollPane(textArea).apply {
            horizontalScrollBarPolicy = JScrollPane.HORIZONTAL_SCROLLBAR_AS_NEEDED
            verticalScrollBarPolicy = JScrollPane.VERTICAL_SCROLLBAR_ALWAYS
        }

        // Filter toolbar
        val filterToolbar = JToolBar().apply {
            isFloatable = false
            add(JLabel(" Filter: "))
        }

        filterCombo = JComboBox(arrayOf("ALL", "GAME", "EMU", "WARN", "ERROR")).apply {
            addActionListener {
                currentFilter = when (selectedItem as String) {
                    "ALL" -> null
                    "GAME" -> LogLevel.GAME
                    "EMU" -> LogLevel.EMU
                    "WARN" -> LogLevel.WARN
                    "ERROR" -> LogLevel.ERROR
                    else -> null
                }
                refreshDisplay()
            }
        }
        filterToolbar.add(filterCombo)

        clearButton = JButton("Clear").apply {
            addActionListener {
                entries.clear()
                textArea.text = ""
            }
        }
        filterToolbar.add(clearButton)

        // Entry count label
        val countLabel = JLabel(" 0 entries ")
        filterToolbar.add(Box.createHorizontalGlue())
        filterToolbar.add(countLabel)

        add(filterToolbar, BorderLayout.NORTH)
        add(scrollPane, BorderLayout.CENTER)

        preferredSize = Dimension(700, 300)
    }

    /**
     * Appends a new log entry. Must be called on EDT or via invokeLater.
     * Color-codes entries by level:
     *   GAME = green, EMU = cyan, WARN = yellow, ERROR = red
     */
    fun appendEntry(entry: DebugLogEntry) {
        SwingUtilities.invokeLater {
            entries.add(entry)
            if (currentFilter == null || entry.level == currentFilter) {
                textArea.append(colorPrefix(entry.level) + entry.formatted())
                // Auto-scroll to bottom
                textArea.caretPosition = textArea.document.length
            }
            // Update count in toolbar
            // (Update the count label)
        }
    }

    private fun refreshDisplay() {
        textArea.text = ""
        val filtered = if (currentFilter == null) entries else entries.filter { it.level == currentFilter }
        for (entry in filtered) {
            textArea.append(colorPrefix(entry.level) + entry.formatted())
        }
        if (textArea.document.length > 0) {
            textArea.caretPosition = textArea.document.length
        }
    }

    // Since JTextArea doesn't support styled text per line easily,
    // use a simpler approach: prepend level tag to each line
    private fun colorPrefix(level: LogLevel): String = when (level) {
        LogLevel.GAME -> "[GAME] "
        LogLevel.EMU -> " [EMU] "
        LogLevel.WARN -> "[WARN] "
        LogLevel.ERROR -> " [ERR] "
    }
}
```

Note: For proper color-coded text, could use JTextPane with StyledDocument instead of JTextArea. However, JTextArea is simpler and the level prefix provides sufficient visual distinction. Use JTextPane if there is time; JTextArea with prefixes is the minimum viable approach.

Also create a convenience class `LogCatWindow` that wraps LogCatPanel in a JFrame:

    }
}
