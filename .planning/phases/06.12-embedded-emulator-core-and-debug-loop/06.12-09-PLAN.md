   ```kotlin
   class EmulatorSession(private val config: EmulatorConfig) {
       private var emulator: CoffeeGbEmulator? = null
       private var window: EmulatorWindow? = null
       private var logWindow: LogCatWindow? = null
       private var memoryWindow: MemoryInspectorWindow? = null
       private var logWriter: DebugLogWriter? = null
       private var sourceMapResolver: SourceMapResolver? = null

       /**
        * Launches the emulator session. Non-blocking — returns immediately.
        *
        * In GUI mode: creates EmulatorWindow with display, toolbar, and
        * optional log/memory inspector windows.
        *
        * In headless mode: starts emulator with log file output only.
        */
       fun launch() {
           // Initialize source map resolver
           sourceMapResolver = SourceMapResolver(config.sourceMapsDir)

           // Initialize log writer
           if (config.logFile != null) {
               logWriter = DebugLogWriter(config.logFile)
           }

           // Create emulator
           val emu = CoffeeGbEmulator(config)
           this.emulator = emu

           // Wire debug entry callback (enriches with source map data)
           emu.onDebugEntry = { entry ->
               // Enrich with source map data
               val enriched = if (entry.cLine != null && sourceMapResolver != null) {
                   val loc = sourceMapResolver!!.resolve(entry.cLine)
                   if (loc != null) {
                       entry.copy(
                           kotlinFile = loc.file,
                           kotlinLine = loc.line,
                           context = loc.context.ifBlank { null }
                       )
                   } else entry
               } else entry

               // Write to log file
               logWriter?.write(enriched)

               // Forward to LogCat panel (if GUI mode)
               logWindow?.logPanel?.let { panel ->
                   SwingUtilities.invokeLater { panel.appendEntry(enriched) }
               }
           }

           if (config.headless) {
               // Headless mode: just start the emulator
               emu.start()
           } else {
               // GUI mode: create all windows
               SwingUtilities.invokeAndWait {
                   // Main emulator window
                   val win = EmulatorWindow(emu, config)
                   this.window = win

                   // Wire frame-ready callback
                   emu.onFrameReady = { frameData ->
                       win.onFrameReady(frameData)
                   }

                   // Create LogCat window (hidden by default, toggled from toolbar)
                   val logWin = LogCatWindow()
                   this.logWindow = logWin

                   // Create Memory Inspector window (hidden by default)
                   val memWin = MemoryInspectorWindow(
                       memoryAccessProvider = { emu.getMemory() },
                       symbolFile = findSymFile()
                   )
                   this.memoryWindow = memWin

                   // Wire toolbar toggles
                   win.toolbar.onLogViewerToggle = { show ->
                       logWin.isVisible = show
                       if (show) logWin.toFront()
                   }
                   win.toolbar.onMemoryInspectorToggle = { show ->
                       memWin.isVisible = show
                       if (show) {
                           memWin.inspectorPanel.refresh()
                           memWin.toFront()
                       }
                   }

                   // Show main window
                   win.isVisible = true
                   win.startFpsTimer()
               }

               // Start emulator (after UI is set up)
               emu.start()
           }
       }

       fun shutdown() {
           emulator?.stop()
           logWriter?.close()
           SwingUtilities.invokeLater {
               window?.dispose()
               logWindow?.dispose()
               memoryWindow?.dispose()
           }
       }

       private fun findSymFile(): File? {
           // Look for .sym file alongside the ROM
           val symFile = File(config.romFile.parentFile, config.romFile.nameWithoutExtension + ".sym")
           return if (symFile.exists()) symFile else null
       }
   }
   ```
