1. Create `GbEmulator.kt` in `io.github.gbkt.emulator` package:
   ```kotlin
   interface GbEmulator {
       fun start()
       fun stop()
       fun pause()
       fun resume()
       fun stepFrame()
       fun setSpeed(multiplier: Float)
       fun getFrameBuffer(): IntArray  // 160*144 RGB pixels
       fun getMemory(): MemoryAccess   // Read/write to emulator address space
       fun getDebugLog(): List<DebugLogEntry>
       fun isRunning(): Boolean
       val isHeadless: Boolean
   }

   /** Wrapper for emulator memory access (reads/writes to the Game Boy address space). */
   interface MemoryAccess {
       fun readByte(address: Int): Int
       fun writeByte(address: Int, value: Int)
   }
   ```
   Also define `LogLevel` enum in same file: `GAME, EMU, WARN, ERROR`

2. Create `EmulatorConfig.kt`:
   ```kotlin
   data class EmulatorConfig(
       val romFile: File,
       val headless: Boolean = false,
       val scale: Int = 4,
       val sourceMapsDir: File? = null,
       val logFile: File? = null,
       val maxLogEntries: Int = 10_000  // bounded circular buffer in memory
   )
   ```

3. Create `DebugLogEntry.kt` in `debug/` subpackage:
   ```kotlin
   data class DebugLogEntry(
       val timestampMs: Long,
       val level: LogLevel,
       val message: String,
       val cLine: Int? = null,
       val kotlinFile: String? = null,
       val kotlinLine: Int? = null,
       val context: String? = null
   ) {
       fun formatted(): String {
           val ts = "[%02d:%02d.%03d]".format(
               timestampMs / 60000, (timestampMs / 1000) % 60, timestampMs % 1000
           )
           val location = if (kotlinFile != null) " $kotlinFile:$kotlinLine" else ""
           val ctx = if (context != null) " ($context)" else ""
           return "$ts$location$ctx > $message\n"
       }
   }
   ```

   ```kotlin
       fun writeByte(address: Int, value: Int)
   }
   ```
