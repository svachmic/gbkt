   ```kotlin
   class GbDisplayPanel(private val scale: Int = 4) : JPanel() {
       private val img = BufferedImage(160, 144, BufferedImage.TYPE_INT_RGB)
       private val pixelBuffer = IntArray(160 * 144)
       private val lock = Object()

       init {
           preferredSize = Dimension(160 * scale, 144 * scale)
           minimumSize = Dimension(160, 144)
       }

       /**
        * Called from the emulator's frame-ready callback.
        * Thread-safe: copies pixel data and schedules repaint on EDT.
        */
       fun onFrame(frameData: IntArray) {
           synchronized(lock) {
               System.arraycopy(frameData, 0, pixelBuffer, 0, minOf(frameData.size, pixelBuffer.size))
           }
           SwingUtilities.invokeLater { repaint() }
       }

       override fun paintComponent(g: Graphics) {
           super.paintComponent(g)
           synchronized(lock) {
               img.setRGB(0, 0, 160, 144, pixelBuffer, 0, 160)
           }
           val g2 = g as Graphics2D
           g2.setRenderingHint(
               RenderingHints.KEY_INTERPOLATION,
               RenderingHints.VALUE_INTERPOLATION_NEAREST_NEIGHBOR
           )
           // Scale to fill panel while maintaining aspect ratio
           val panelW = width
           val panelH = height
           val scaleX = panelW.toDouble() / 160.0
           val scaleY = panelH.toDouble() / 144.0
           val actualScale = minOf(scaleX, scaleY)
           val drawW = (160 * actualScale).toInt()
           val drawH = (144 * actualScale).toInt()
           val offsetX = (panelW - drawW) / 2
           val offsetY = (panelH - drawH) / 2
           g2.color = Color.BLACK
           g2.fillRect(0, 0, panelW, panelH)
           g2.drawImage(img, offsetX, offsetY, drawW, drawH, null)
       }
   }
   ```

2. **Create InputHandler.kt:**
   ```kotlin
   class InputHandler(private val controller: /* Coffee-GB Controller or Joypad type */) : KeyListener {

       // Mapping from CONTEXT.md:
       // Arrow keys = D-pad, Z = A, X = B, Enter = Start, Backspace = Select
       private fun mapKey(keyCode: Int): Button? = when (keyCode) {
           KeyEvent.VK_UP -> Button.UP
           KeyEvent.VK_DOWN -> Button.DOWN
           KeyEvent.VK_LEFT -> Button.LEFT
           KeyEvent.VK_RIGHT -> Button.RIGHT
           KeyEvent.VK_Z -> Button.A
           KeyEvent.VK_X -> Button.B
           KeyEvent.VK_ENTER -> Button.START
           KeyEvent.VK_BACK_SPACE -> Button.SELECT
           else -> null
       }

       override fun keyPressed(e: KeyEvent) {
           val button = mapKey(e.keyCode) ?: return
           controller.buttonPressed(button)  // Or post ButtonPressEvent to EventBus
       }

       override fun keyReleased(e: KeyEvent) {
           val button = mapKey(e.keyCode) ?: return
           controller.buttonReleased(button)  // Or post ButtonReleaseEvent to EventBus
       }

       override fun keyTyped(e: KeyEvent) {} // Not used
   }
   ```

   IMPORTANT: Check Coffee-GB's actual input API. It may use:

   - OR `EventBus` posts `ButtonPressEvent` / `ButtonReleaseEvent`
   - OR `Controller` interface
   Adapt the InputHandler to match the actual API available.
