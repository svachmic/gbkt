   class NamedVariablesTab(
       private val addressSpaceProvider: () -> AddressSpace?,
       private val symbolFile: File? = null  // .sym file from lcc build
   ) : JPanel(BorderLayout()) {
       private val tableModel: DefaultTableModel
       private val table: JTable
       private val variables = mutableListOf<VariableEntry>()

       data class VariableEntry(
           val name: String,
           val type: String,  // UINT8, INT8, UINT16, etc.
           val address: Int,
           var value: String = "?"
       )

       init {
           tableModel = DefaultTableModel(
               arrayOf("Name", "Type", "Address", "Value"),
               0
           )
           table = JTable(tableModel).apply {
               font = Font(Font.MONOSPACED, Font.PLAIN, 12)
               setSelectionMode(ListSelectionModel.SINGLE_SELECTION)
           }

           // Load variables from .sym file if available
           if (symbolFile != null && symbolFile.exists()) {
               loadSymbols(symbolFile)
           }

           add(JScrollPane(table), BorderLayout.CENTER)

           // Refresh button
           val toolbar = JToolBar().apply {
               isFloatable = false
               add(JButton("Refresh").apply {
                   addActionListener { refresh() }
               })
               add(JButton("Add Variable...").apply {
                   addActionListener { addManualVariable() }
               })
           }
           add(toolbar, BorderLayout.NORTH)
       }

       fun refresh() {
           val addressSpace = addressSpaceProvider() ?: return
           for (i in variables.indices) {
               val v = variables[i]
               val rawValue = addressSpace.getByte(v.address) and 0xFF
               v.value = when (v.type) {
                   "UINT8" -> "$rawValue"
                   "INT8" -> "${rawValue.toByte()}"
                   "UINT16" -> {
                       val lo = rawValue
                       val hi = addressSpace.getByte(v.address + 1) and 0xFF
                       "${(hi shl 8) or lo}"  // Little-endian
                   }
                   else -> "0x${rawValue.toString(16).padStart(2, '0')}"
               }
               tableModel.setValueAt(v.value, i, 3)
           }
       }

       private fun loadSymbols(symFile: File) {
           // Parse SDCC .sym file format:
           // DEF symbolName addr
           // Lines like: "DEF _score 00:C100"
           symFile.readLines().forEach { line ->
               val parts = line.trim().split("\\s+".toRegex())
               if (parts.size >= 3 && parts[0] == "DEF" && parts[1].startsWith("_")) {
                   val name = parts[1].removePrefix("_")
                   val addrStr = parts[2]
                   val address = try {
                       // Format: "00:C100" (bank:address)
                       val addrPart = addrStr.substringAfter(":")
                       addrPart.toInt(16)
                   } catch (e: Exception) { return@forEach }
                   val entry = VariableEntry(name, "UINT8", address)
                   variables.add(entry)
                   tableModel.addRow(arrayOf(name, "UINT8", "0x${address.toString(16).uppercase()}", "?"))
               }
           }
       }

       private fun addManualVariable() {
           // Show dialog to add a variable by name, type, and address
           val name = JOptionPane.showInputDialog(this, "Variable name:") ?: return
           val addressStr = JOptionPane.showInputDialog(this, "Address (hex, e.g. C100):") ?: return
           val address = try { addressStr.toInt(16) } catch (e: Exception) {
               JOptionPane.showMessageDialog(this, "Invalid hex address"); return
           }
           val entry = VariableEntry(name, "UINT8", address)
           variables.add(entry)
           tableModel.addRow(arrayOf(name, "UINT8", "0x${address.toString(16).uppercase()}", "?"))
       }
   }

   class HexViewTab(
       private val addressSpaceProvider: () -> AddressSpace?
   ) : JPanel(BorderLayout()) {
       private val textArea: JTextArea
       private val addressField: JTextField
       private var baseAddress: Int = 0xC000  // Start at WRAM

       init {
           textArea = JTextArea().apply {
               font = Font(Font.MONOSPACED, Font.PLAIN, 12)
               isEditable = false
               background = Color(30, 30, 30)
               foreground = Color(200, 200, 200)
           }

           val toolbar = JToolBar().apply {
               isFloatable = false
               add(JLabel(" Address: 0x"))
               addressField = JTextField("C000", 6).apply {
                   addActionListener { navigateTo() }
               }
               add(addressField)
               add(JButton("Go").apply { addActionListener { navigateTo() } })
               add(JButton("Refresh").apply { addActionListener { refresh() } })

               addSeparator()
               // Quick navigation buttons
               add(JButton("WRAM").apply { addActionListener { goTo(0xC000) } })
               add(JButton("HRAM").apply { addActionListener { goTo(0xFF80) } })
               add(JButton("OAM").apply { addActionListener { goTo(0xFE00) } })
               add(JButton("IO").apply { addActionListener { goTo(0xFF00) } })
           }

           add(toolbar, BorderLayout.NORTH)
           add(JScrollPane(textArea), BorderLayout.CENTER)

           preferredSize = Dimension(600, 400)
       }

       fun refresh() {
           val addressSpace = addressSpaceProvider() ?: return
           val sb = StringBuilder()
           // Show 256 bytes (16 rows of 16 bytes)
           sb.appendLine("Address   00 01 02 03 04 05 06 07  08 09 0A 0B 0C 0D 0E 0F  ASCII")
           sb.appendLine("-".repeat(75))
           for (row in 0 until 16) {
               val addr = baseAddress + row * 16
               sb.append("${addr.toString(16).uppercase().padStart(4, '0')}:  ")
               val ascii = StringBuilder()
               for (col in 0 until 16) {
                   val byte = addressSpace.getByte(addr + col) and 0xFF
                   sb.append(byte.toString(16).uppercase().padStart(2, '0'))
                   sb.append(if (col == 7) "  " else " ")
                   ascii.append(if (byte in 0x20..0x7E) byte.toChar() else '.')
               }
               sb.append(" $ascii")
               sb.appendLine()
           }
           textArea.text = sb.toString()
       }

       private fun navigateTo() {
           val addr = try { addressField.text.toInt(16) } catch (e: Exception) {
               JOptionPane.showMessageDialog(this, "Invalid hex address"); return
           }
           goTo(addr)
       }

       private fun goTo(address: Int) {
           baseAddress = address
           addressField.text = address.toString(16).uppercase()
           refresh()
       }
   }
