```kotlin
class EmulatorToolbar(private val emulator: GbEmulator) : JToolBar() {
    private val pauseResumeButton: JButton
    private val stepButton: JButton
    private val speedGroup: ButtonGroup
    private val speed1x: JToggleButton
    private val speed2x: JToggleButton
    private val speed4x: JToggleButton
    private val memoryButton: JToggleButton

    var onMemoryInspectorToggle: ((Boolean) -> Unit)? = null
    var onLogViewerToggle: ((Boolean) -> Unit)? = null

    init {
        isFloatable = false

        // Pause/Resume button
        pauseResumeButton = JButton("Pause").apply {
            toolTipText = "Pause/Resume emulation (Space)"
            addActionListener { togglePause() }
        }
        add(pauseResumeButton)

        // Step Frame button
        stepButton = JButton("Step").apply {
            toolTipText = "Advance one frame (F10)"
            isEnabled = false  // Only enabled when paused
            addActionListener {
                emulator.stepFrame()
            }
        }
        add(stepButton)

        addSeparator()

        // Speed buttons
        add(JLabel("Speed: "))
        speedGroup = ButtonGroup()
        speed1x = JToggleButton("1x", true).apply {
            addActionListener { emulator.setSpeed(1.0f) }
        }
        speed2x = JToggleButton("2x").apply {
            addActionListener { emulator.setSpeed(2.0f) }
        }
        speed4x = JToggleButton("4x").apply {
            addActionListener { emulator.setSpeed(4.0f) }
        }
        speedGroup.add(speed1x)
        speedGroup.add(speed2x)
        speedGroup.add(speed4x)
        add(speed1x)
        add(speed2x)
        add(speed4x)

        addSeparator()

        // Log Viewer toggle
        val logButton = JToggleButton("Log").apply {
            toolTipText = "Toggle debug log viewer"
            addActionListener { onLogViewerToggle?.invoke(isSelected) }
        }
        add(logButton)

        // Memory Inspector toggle
        memoryButton = JToggleButton("Memory").apply {
            toolTipText = "Toggle memory inspector"
            addActionListener { onMemoryInspectorToggle?.invoke(isSelected) }
        }
        add(memoryButton)
    }

    private fun togglePause() {
        if (emulator.isRunning()) {
            emulator.pause()
            pauseResumeButton.text = "Resume"
            stepButton.isEnabled = true
        } else {
            emulator.resume()
            pauseResumeButton.text = "Pause"
            stepButton.isEnabled = false
        }
    }

    fun updatePauseState(paused: Boolean) {
        SwingUtilities.invokeLater {
            pauseResumeButton.text = if (paused) "Resume" else "Pause"
            stepButton.isEnabled = paused
        }
    }
}
```

   class EmulatorWindow(
       private val emulator: GbEmulator,
       private val config: EmulatorConfig
   ) : JFrame() {
       val displayPanel: GbDisplayPanel
       private val toolbar: EmulatorToolbar
       private val statusBar: JLabel
       private var frameCount: Long = 0
       private var fpsTimer: Timer? = null
       private var lastFrameCount: Long = 0

       init {
           val romName = config.romFile.nameWithoutExtension
           title = "gbkt - $romName | 0.0 FPS | Frame 0"
           defaultCloseOperation = DO_NOTHING_ON_CLOSE
           addWindowListener(object : java.awt.event.WindowAdapter() {
               override fun windowClosing(e: java.awt.event.WindowEvent) {
                   shutdown()
               }
           })

           // Display panel (center)
           displayPanel = GbDisplayPanel(config.scale)

           // Toolbar (top)
           toolbar = EmulatorToolbar(emulator)
           toolbar.onLogViewerToggle = { show -> toggleLogViewer(show) }
           toolbar.onMemoryInspectorToggle = { show -> toggleMemoryInspector(show) }

           // Status bar (bottom)
           statusBar = JLabel(" Ready").apply {
               border = BorderFactory.createEtchedBorder()
           }

           // Layout
           layout = BorderLayout()
           add(toolbar, BorderLayout.NORTH)
           add(displayPanel, BorderLayout.CENTER)
           add(statusBar, BorderLayout.SOUTH)

           // Input handling - add to frame for focus
           addKeyListener(InputHandler(/* controller from emulator */))
           isFocusable = true

           pack()
           setLocationRelativeTo(null)  // Center on screen
           isResizable = true
       }

       fun startFpsTimer() {
           fpsTimer = Timer(1000) {  // Update every second
               SwingUtilities.invokeLater {
                   val fps = frameCount - lastFrameCount
                   lastFrameCount = frameCount
                   val romName = config.romFile.nameWithoutExtension
                   title = "gbkt - $romName | $fps FPS | Frame $frameCount"
               }
           }
           fpsTimer?.start()
       }

       fun onFrameReady(frameData: IntArray) {
           frameCount++
           displayPanel.onFrame(frameData)
       }

       private fun shutdown() {
           fpsTimer?.stop()
           emulator.stop()
           dispose()
       }
