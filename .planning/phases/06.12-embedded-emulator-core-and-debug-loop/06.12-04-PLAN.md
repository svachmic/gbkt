   ```kotlin
   class SourceMapResolver(sourceMapsDir: File?) {
       data class SourceLocation(
           val file: String,
           val line: Int,
           val symbol: String = "",
           val context: String = ""  // e.g. "gameplay/frame" from snippet
       )

       private val mappings: Map<Int, SourceLocation>  // cLine → location

       init {
           mappings = if (sourceMapsDir != null && sourceMapsDir.exists()) {
               loadMaps(sourceMapsDir)
           } else {
               emptyMap()
           }
       }

       fun resolve(cLine: Int): SourceLocation? = mappings[cLine]

       private fun loadMaps(dir: File): Map<Int, SourceLocation> {
           val result = mutableMapOf<Int, SourceLocation>()
           dir.listFiles { f -> f.name.endsWith(".gbkt.map") }?.forEach { mapFile ->
               val json = JSONObject(mapFile.readText())
               val mappingsArray = json.getJSONArray("mappings")
               for (i in 0 until mappingsArray.length()) {
                   val m = mappingsArray.getJSONObject(i)
                   val cLine = m.getInt("cLine")
                   result[cLine] = SourceLocation(
                       file = m.getString("kotlinFile"),
                       line = m.getInt("kotlinLine"),
                       symbol = m.optString("symbol", ""),
                       context = m.optString("snippet", "")
                   )
               }
           }
           return result
       }
   }
   ```

3. **Create DebugLogWriter.kt:**
   ```kotlin
   class DebugLogWriter(logFile: File) : Closeable {
       private val writer: PrintWriter

       init {
           logFile.parentFile?.mkdirs()
           // Overwrite on each run (fresh per invocation per CONTEXT.md)
           writer = PrintWriter(BufferedWriter(FileWriter(logFile, false)), true)  // autoFlush=true
       }

       fun write(entry: DebugLogEntry) {
           writer.print(entry.formatted())
           writer.flush()  // Ensure real-time visibility for tail -f
       }

       override fun close() {
           writer.close()
       }
   }
   ```
