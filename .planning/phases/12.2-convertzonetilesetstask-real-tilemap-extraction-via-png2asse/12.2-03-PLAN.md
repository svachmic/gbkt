    /**
     * Parse a `const unsigned char <symbol>[N] = { 0x.., 0x.., ... };` byte-array initializer
     * from a png2asset-emitted .c file. Returns the byte values as `List<Int>` (each in 0..255),
     * or `null` if the symbol is not found.
     *
     * Phase 12.2 D-02: shared parse step for both invocation paths.
     */
    private fun parseMapArrayBytes(file: File, symbol: String): List<Int>? {
        val text = file.readText()
        val pattern = Regex(
            """const\s+unsigned\s+char\s+${Regex.escape(symbol)}\s*\[\s*\d+\s*\]\s*=\s*\{([^}]*)\}""",
            RegexOption.DOT_MATCHES_ALL,
        )
        val match = pattern.find(text) ?: return null
        val payload = match.groupValues[1]
        return payload
            .split(',')
            .map { it.trim() }
            .filter { it.isNotEmpty() }
            .map { token -> token.removePrefix("0x").removePrefix("0X").toInt(16) }
    }
