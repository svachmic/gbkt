       ```kotlin
       class PhysicsBuilder {
           private var vx: Int = 0
           private var vy: Int = 0
           private var ax: Int = 0
           private var ay: Int = 0
           private var grav: Int = 0
           private var bounceCoeff: Int = 0
           private var maxFall: Int = 8

           fun velocity(dx: Int, dy: Int) { vx = dx; vy = dy }
           fun acceleration(ax: Int, ay: Int) { this.ax = ax; this.ay = ay }
           fun gravity(pixelsPerFrameSq: Int) { grav = pixelsPerFrameSq }
           fun bounce(coefficient: Float) { bounceCoeff = (coefficient * 256).toInt().coerceIn(0, 255) }
           fun maxFallSpeed(speed: Int) { maxFall = speed }

           internal fun build(): PhysicsConfig = PhysicsConfig(vx, vy, ax, ay, grav, bounceCoeff, maxFall)
       }
       ```

       fun physics(block: PhysicsBuilder.() -> Unit) {

       // Convenience methods:

       fun gravity(g: Int) { ensurePhysics().also { it.grav = g } }
       fun bounce(coeff: Float) { ensurePhysics().also { it.bounceCoeff = (coeff * 256).toInt() } }

    2. In ScriptOpVisitor.kt, implement `visitPhysicsStep(op)`:
       Generate typed C AST for per-frame physics:
       ```kotlin
       CBlock(listOf(
           // Apply acceleration to velocity: _player_vx += ACCEL_X_PLAYER; _player_vy += ACCEL_Y_PLAYER
           CExprStatement(CBinaryExpr(CVar("_${actorId}_vx"), "+=", CVar("ACCEL_X_${actorIdUpper}"))),
           CExprStatement(CBinaryExpr(CVar("_${actorId}_vy"), "+=", CVar("ACCEL_Y_${actorIdUpper}"))),
           // Apply gravity to VY: _player_vy += GRAVITY_PLAYER
           CExprStatement(CBinaryExpr(CVar("_${actorId}_vy"), "+=", CVar("GRAVITY_${actorIdUpper}"))),
           // Clamp fall speed: if (_player_vy > MAX_FALL_PLAYER) _player_vy = MAX_FALL_PLAYER
           CIf(CBinaryExpr(CVar("_${actorId}_vy"), ">", CVar("MAX_FALL_${actorIdUpper}")),
               CExprStatement(CBinaryExpr(CVar("_${actorId}_vy"), "=", CVar("MAX_FALL_${actorIdUpper}")))),
           // Apply VY to Y: _player_y += _player_vy
           CExprStatement(CBinaryExpr(CVar("_${actorId}_y"), "+=", CCast(CU8, CVar("_${actorId}_vy")))),
           // Apply VX to X: _player_x += _player_vx
           CExprStatement(CBinaryExpr(CVar("_${actorId}_x"), "+=", CCast(CU8, CVar("_${actorId}_vx")))),
           // Floor bounce: if (_player_y >= FLOOR_Y) { _player_y = FLOOR_Y; _player_vy = -(_player_vy * BOUNCE / 256); }
           // Note: floor detection is game-specific (collision) — generate generic bounce that game can call
       ))
       ```
       Note: Floor collision is scene-specific. PhysicsStep applies acceleration + gravity + velocity only. Bounce detection is handled by the game's whenever() conditions. Document this: PhysicsStep does acceleration+gravity+velocity; bounce/floor is game logic.
