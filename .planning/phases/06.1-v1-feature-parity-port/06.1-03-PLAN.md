Add per-actor movement configuration and sprite animation state machines to the V2 pipeline.        class MovementBuilder {
           private var style: MovementStyle = MovementStyle.GRID
           private var speed: Int = 4
           private var tileSize: Int = 8

           fun style(s: MovementStyle) { style = s }
           fun speed(pixelsPerStep: Int) { speed = pixelsPerStep }
           fun tileSize(px: Int) { tileSize = px }

           internal fun build(): MovementConfig = MovementConfig(style, speed, tileSize)
       }

       class AnimationStateBuilder(private val name: String) {
           private var startFrame: Int = 0
           private var endFrame: Int = 0
           private var speed: Int = 8
           private var loop: Boolean = true
           private val transitions: MutableList<AnimTransition> = mutableListOf()

           fun frames(range: IntRange) { startFrame = range.first; endFrame = range.last }
           fun speed(framesPerUpdate: Int) { speed = framesPerUpdate }
           fun loop(enabled: Boolean) { loop = enabled }

           internal fun build(): AnimationStateDef = AnimationStateDef(name, startFrame, endFrame, speed, loop, transitions.toList())
       }

       class AnimationStatesBuilder {
           private val states: MutableList<AnimationStateDef> = mutableListOf()
           private val transitions: MutableList<AnimTransition> = mutableListOf()

           fun state(name: String, block: AnimationStateBuilder.() -> Unit) {
               states.add(AnimationStateBuilder(name).apply(block).build())
           }

           /** Condition-based auto-transition: transition("idle" to "walk") { dpad.any } */
           fun transition(pair: Pair<String, String>, conditionBlock: () -> Expr) {
               val condition = conditionBlock().toIR()  // convert DSL Expr to IRExpression
               transitions.add(AnimTransition(fromState = pair.first, toState = pair.second, condition = condition))
           }

           internal fun build(): List<AnimationStateDef> {
               // Attach transitions to their fromState's AnimationStateDef
               return states.map { state ->
                   val stateTransitions = transitions.filter { it.fromState == state.name }
                   state.copy(transitions = state.transitions + stateTransitions)
               }
           }
       }
