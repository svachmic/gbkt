    ```kotlin
    class CurrencyBuilder(private val id: String) {
        private var max: Int = 9999
        private val exchanges = mutableListOf<CurrencyExchange>()
        fun max(value: Int) { max = value }
        fun exchange(to: CurrencyRef, rate: Int) { exchanges.add(CurrencyExchange(to.id, rate)) }
        fun build(): CurrencyDef = CurrencyDef(id, max, exchanges)
    }
    class CurrencyDelegate(private val block: CurrencyBuilder.() -> Unit) {
        operator fun provideDelegate(thisRef: Any?, prop: KProperty<*>): ReadOnlyProperty<Any?, CurrencyRef> { ... }
    }

    ```
