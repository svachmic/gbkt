**HashTable C output (static linear probing):**

Data:
```c
// === Hash Table: tile_cache ===
#define TILE_CACHE_SIZE 64
static UINT16 tile_cache_keys[64];
static UINT8 tile_cache_values[64];
static UINT8 tile_cache_used[64];
```

Functions:
```c
static UINT8 tile_cache_hash(UINT16 key) {
    return (UINT8)(key % TILE_CACHE_SIZE);
}

static void tile_cache_insert(UINT16 key, UINT8 value) {
    UINT8 idx = tile_cache_hash(key);
    UINT8 i;
    for (i = 0; i < TILE_CACHE_SIZE; i++) {
        UINT8 probe = (idx + i) % TILE_CACHE_SIZE;
        if (!tile_cache_used[probe] || tile_cache_keys[probe] == key) {
            tile_cache_keys[probe] = key;
            tile_cache_values[probe] = value;
            tile_cache_used[probe] = 1;
            return;
        }
    }
}

static UINT8 tile_cache_lookup(UINT16 key, UINT8 default_val) {
    UINT8 idx = tile_cache_hash(key);
    UINT8 i;
    for (i = 0; i < TILE_CACHE_SIZE; i++) {
        UINT8 probe = (idx + i) % TILE_CACHE_SIZE;
        if (!tile_cache_used[probe]) return default_val;
        if (tile_cache_keys[probe] == key) return tile_cache_values[probe];
    }
    return default_val;
}

static void tile_cache_clear(void) {
    UINT8 i;
    for (i = 0; i < TILE_CACHE_SIZE; i++) {
        tile_cache_used[i] = 0;
    }
}
```

**Pool C output (free-list bitmap):**

Data:
```c
#define ITEM_POOL_CAPACITY 16
static UINT8 item_pool_data[16];
static UINT8 item_pool_free[16];
static UINT8 item_pool_count;
```

Functions:
```c
static void item_pool_init(void) {
    UINT8 i;
    item_pool_count = 0;
    for (i = 0; i < ITEM_POOL_CAPACITY; i++) {
        item_pool_free[i] = 1;
    }
}

static INT8 item_pool_alloc(void) {
    UINT8 i;
    if (item_pool_count >= ITEM_POOL_CAPACITY) return -1;
    for (i = 0; i < ITEM_POOL_CAPACITY; i++) {
        if (item_pool_free[i]) {
            item_pool_free[i] = 0;
            item_pool_count++;
            return (INT8)i;
        }
    }
    return -1;
}

static void item_pool_free_slot(UINT8 idx) {
    if (idx < ITEM_POOL_CAPACITY && !item_pool_free[idx]) {
        item_pool_free[idx] = 1;
        item_pool_count--;
    }
}
```

**RingBuffer C output (head/tail indices):**

Data:
```c
#define MSG_BUFFER_CAPACITY 128
static UINT8 msg_buffer_data[128];
static UINT8 msg_buffer_head;
static UINT8 msg_buffer_tail;
static UINT8 msg_buffer_count;
```

Functions:
```c
static void msg_buffer_push(UINT8 val) {
    if (msg_buffer_count >= MSG_BUFFER_CAPACITY) return;
    msg_buffer_data[msg_buffer_tail] = val;
    msg_buffer_tail = (msg_buffer_tail + 1) % MSG_BUFFER_CAPACITY;
    msg_buffer_count++;
}

static UINT8 msg_buffer_pop(void) {
    UINT8 val;
    if (msg_buffer_count == 0) return 0;
    val = msg_buffer_data[msg_buffer_head];
    msg_buffer_head = (msg_buffer_head + 1) % MSG_BUFFER_CAPACITY;
    msg_buffer_count--;
    return val;
}

static UINT8 msg_buffer_peek(void) {
    if (msg_buffer_count == 0) return 0;
    return msg_buffer_data[msg_buffer_head];
}
```

**FixedSlots C output (bitfield active tracking):**

Data:
```c
#define ENEMY_SLOTS_COUNT 3
static UINT16 enemy_slots_data[3];
static UINT8 enemy_slots_active;  // UINT8 for count<=8, UINT16 for count 9-16
```

Functions:
```c
static INT8 enemy_slots_claim(void) {
    UINT8 i;
    for (i = 0; i < ENEMY_SLOTS_COUNT; i++) {
        if (!(enemy_slots_active & (1 << i))) {
            enemy_slots_active |= (1 << i);
            return (INT8)i;
        }
    }
    return -1;
}

static void enemy_slots_release(UINT8 idx) {
    if (idx < ENEMY_SLOTS_COUNT) {
        enemy_slots_active &= ~(1 << idx);
    }
}

static UINT8 enemy_slots_is_active(UINT8 idx) {
    return (enemy_slots_active & (1 << idx)) ? 1 : 0;
}
```
