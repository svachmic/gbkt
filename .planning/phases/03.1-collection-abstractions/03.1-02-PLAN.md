**1. HashTableCollection.kt:**
```kotlin
class HashTableCollection(
    val name: String,
    val keyType: GBVar.VarType,
    val valueType: GBVar.VarType,
    val size: Int,
) {
    /** Emit insert operation */
    fun insert(key: Expr, value: Expr) {
        RecordingContext.require().emit(IRCollHashTableInsert(name, key.unwrap(), value.unwrap()))
    }

    /** Emit clear operation */
    fun clear() {
        RecordingContext.require().emit(IRCollHashTableClear(name))
    }

    /** Expression: look up key, returns value or 0xFF if not found */
    fun lookup(key: Expr): Expr = Expr(IRCollHashTableLookup(name, key.unwrap()))

    /** IR descriptor for this collection */
    fun toIR(): IRCollHashTable = IRCollHashTable(name, keyType, valueType, size)

    /** Compile-time byte count */
    val sizeBytes: Int get() = toIR().sizeBytes
}
```

NOTE: `Expr` is the DSL-facing wrapper around `IRExpression` from ExpressionWrapper.kt. Use `Expr(irExpr)` to wrap and `expr.unwrap()` (or however the existing pattern extracts the inner IRExpression) to unwrap. Check ExpressionWrapper.kt for the exact accessor name.

**CollectionDelegates.kt** — Create DSL entry-point functions following the U8ArrayDelegate pattern from Variables.kt:

```kotlin
class HashTableDelegate(
    private val keyType: GBVar.VarType,
    private val valueType: GBVar.VarType,
    private val size: Int,
) {
    private var table: HashTableCollection? = null

    operator fun getValue(thisRef: Any?, property: KProperty<*>): HashTableCollection {
        if (table == null) {
            val t = HashTableCollection(property.name, keyType, valueType, size)
            table = t
            // Register with GameScope so GameBuilder.build() picks it up
            GameScopeContext.current?.registerHashTable(t.toIR())
        }
        return table!!
    }
}

// Top-level DSL functions
fun hashtable(keyType: GBVar.VarType, valueType: GBVar.VarType, size: Int) =
    HashTableDelegate(keyType, valueType, size)

// Convenience overloads for common patterns:
// val tileCache by hashtable(VarType.U16, VarType.U8, 64)
```

Repeat for the other 3 types:

The generic `<T>` sugar can be added in a future phase when struct element types are supported.
