**PongV2.kt** — Pong using new DSL:
```kotlin
val pongV2 = game("Pong") {
    config { cartridge = "ROM_ONLY"; romBanks = 2 }

    // Variables
    var p1Score by u8Var(0)
    var p2Score by u8Var(0)
    var ballDx by i8Var(1)
    var ballDy by i8Var(1)

    // Actors
    val paddle1 = actor("paddle1") {
        position(16, 64)
        sprite(asset("sprites/paddle.png")) { size(8, 16); hitbox(0, 0, 8, 16) }
    }
    val paddle2 = actor("paddle2") { ... }
    val ball = actor("ball") {
        position(80, 72)
        sprite(asset("sprites/ball.png")) { size(8, 8); hitbox(0, 0, 8, 8) }
    }

    // Title scene
    scene("title") {
        enter { hideSprites(); clear(); print("PONG", position = center(6)); print("PRESS START", position = center(10)) }
        frame { whenever(buttonPressed("start")) { navigate("game") } }
    }

    // Game scene
    scene("game") {
        enter { showSprites(); setPosition(ball, 80, 72); assign("ballDx", literal(1)) ... }
        frame {
            // P1 controls
            whenever(dpadHeld("up")) { moveBy(paddle1, 0, -2) }
            whenever(dpadHeld("down")) { moveBy(paddle1, 0, 2) }
            // AI for P2
            whenever(varRef("paddle2.y") isAbove varRef("ball.y")) { moveBy(paddle2, 0, -1) }
            whenever(varRef("paddle2.y") isBelow varRef("ball.y")) { moveBy(paddle2, 0, 1) }
            // Ball movement
            assign("ball.x", varRef("ball.x") + varRef("ballDx"))
            assign("ball.y", varRef("ball.y") + varRef("ballDy"))
            // Bouncing
            whenever(varRef("ball.y") isBelow literal(16)) { assign("ballDy", literal(1)) }
            whenever(varRef("ball.y") isAbove literal(148)) { assign("ballDy", literal(-1)) }
            // Collision with paddles
            whenever(collides(ball, paddle1)) { assign("ballDx", literal(1)) }
            whenever(collides(ball, paddle2)) { assign("ballDx", literal(-1)) }
            // Scoring
            whenever(varRef("ball.x") isBelow literal(4)) {
                assign("p2Score", varRef("p2Score") + literal(1))
                setPosition(ball, 80, 72); assign("ballDx", literal(1))
            }
            whenever(varRef("ball.x") isAbove literal(156)) {
                assign("p1Score", varRef("p1Score") + literal(1))
                setPosition(ball, 80, 72); assign("ballDx", literal(-1))
            }
            // Win
            whenever(varRef("p1Score") isAtLeast literal(5)) { navigate("gameover") }
            whenever(varRef("p2Score") isAtLeast literal(5)) { navigate("gameover") }
        }
    }

    // Game over scene
    scene("gameover") {
        enter { hideSprites(); clear(); print("GAME OVER", position = center(7)); print("PRESS START", position = center(14)) }
        frame { whenever(buttonPressed("start")) { assign("p1Score", literal(0)); assign("p2Score", literal(0)); navigate("title") } }
    }

    start = "title"
}
```

val explorerV2 = game("Explorer") {
    config { cartridge = "MBC5_RAM_BATTERY"; romBanks = 4; ramBanks = 1 }

    // Game state
    var roomId by u8Var(0)
    var stepCount by u8Var(0)
    var keys by u8Var(0)
    var torchLevel by u8Var(255)

    // Sound effects
    val stepSound = soundEffect("step") { preset = "TICK" }
    val doorSound = soundEffect("door") { preset = "SELECT" }
    val saveSound = soundEffect("save") { preset = "POWERUP" }
    val hitSound = soundEffect("hit") { preset = "HIT" }

    // Player entity
    val player = actor("player") {
        position(80, 72)
        sprite(asset("sprites/player.png")) { size(8, 16); hitbox(0, 8, 8, 8) }
    }

    // Camera
    camera { smoothing = 0.2f }

    // Save data
    saveData("explorer") {
        slots = 1
    }

    // RPG definitions — using gbkt-rpg DSL builders
    val hero = character("hero") {
        name("Hero")
        stats { hp(20); atk(5); def(3) }
    }

    val goblin = monster("goblin") {
        name("Goblin")
        stats { hp(10); atk(3); def(1) }
        exp(5)
    }

    val rat = monster("rat") {
        name("Rat")
        stats { hp(5); atk(2); def(0) }
        exp(2)
    }

    // Simple battle system — produces SystemIR.GenericSystem (core IR)
    simpleBattle("combat") {
        party(hero)
        encounter { +goblin }
        encounter { +rat }
        onVictory { navigate("gameplay") }
        onDefeat { navigate("gameover") }
    }

    // Title scene
    scene("title") {
        enter { hideSprites(); clear(); print("EXPLORER", position = center(5)); print("PRESS START", position = center(12)) }
        frame { whenever(buttonPressed("start")) { navigate("gameplay") } }
    }

    // Gameplay scene — grid movement, torch gauge, step counter
    scene("gameplay") {
        enter { showSprites(); clear() }
        frame {
            // 4-directional movement
            whenever(dpadHeld("up")) { moveBy(player, 0, -1) }
            whenever(dpadHeld("down")) { moveBy(player, 0, 1) }
            whenever(dpadHeld("left")) { moveBy(player, -1, 0) }
            whenever(dpadHeld("right")) { moveBy(player, 1, 0) }
            // Clamp to screen
            whenever(varRef("player.x") isBelow literal(8)) { setPosition(player, 8, varRef("player.y")) }
            whenever(varRef("player.x") isAbove literal(152)) { setPosition(player, 152, varRef("player.y")) }
            whenever(varRef("player.y") isBelow literal(16)) { setPosition(player, varRef("player.x"), 16) }
            whenever(varRef("player.y") isAbove literal(144)) { setPosition(player, varRef("player.x"), 144) }
            // Torch diminishes
            assign("torchLevel", varRef("torchLevel") - literal(1))
            // HUD
            print("HP:", varRef("hp"), position = at(1, 0))
            print("KEYS:", varRef("keys"), position = at(10, 0))
            print("TORCH:", varRef("torchLevel"), position = at(1, 17))
            // Save
            whenever(buttonPressed("select")) { playSound("save") }
            // Pause
            whenever(buttonPressed("start")) { navigate("pause") }
            // Random encounter trigger (simplified: every 20 steps)
            whenever(varRef("stepCount") isAtLeast literal(20)) {
                assign("stepCount", literal(0))
                navigate("combat_scene")
            }
        }
    }

    // Pause scene
    scene("pause") {
        enter { hideSprites(); clear(); print("PAUSED", position = center(7)); print("START:RESUME", position = center(10)); print("B:TITLE", position = center(12)) }
        frame {
            whenever(buttonPressed("start")) { navigate("gameplay") }
            whenever(buttonPressed("b")) { navigate("title") }
        }
    }

    // Combat scene — uses gbkt-rpg simpleBattle system via battleUpdate
    scene("combat_scene") {
        enter { hideSprites(); clear() }
        frame {
            battleUpdate("combat")  // Drives the simple battle state machine (produces TriggerSystem ScriptOp)
        }
    }

    // Game over scene
    scene("gameover") {
        enter { hideSprites(); clear(); print("GAME OVER", position = center(7)); print("PRESS START", position = center(14)) }
        frame { whenever(buttonPressed("start")) { navigate("title") } }
    }

    start = "title"
}

class ExplorerIRTest {
    private val ir = explorerV2.build()

    @Test fun `has 5 scenes`() = assertEquals(5, ir.scenes.size)  // title, gameplay, pause, combat_scene, gameover
    @Test fun `has 1 actor`() = assertEquals(1, ir.actors.size)
    @Test fun `start scene is title`() = assertEquals("title", ir.startScene)
    @Test fun `has torch variable`() { assertTrue(ir.variables.any { it.name == "torchLevel" }) }
    @Test fun `has camera system`() { assertTrue(ir.systems.any { it is CameraSystem }) }
    @Test fun `has save system`() { assertTrue(ir.systems.any { it is SaveSystem }) }
    @Test fun `has combat system from RPG builder`() {
        // The simpleBattle {} builder registers a GenericSystem with type "simple_battle"
        val combatSystem = ir.systems.filterIsInstance<GenericSystem>()
            .find { it.id == "combat" }
        assertNotNull(combatSystem, "Expected a GenericSystem with id 'combat' from simpleBattle builder")
        assertEquals("simple_battle", combatSystem!!.config["type"])
    }
    @Test fun `combat system contains party and encounter data`() {
        val combatSystem = ir.systems.filterIsInstance<GenericSystem>()
            .find { it.id == "combat" }!!
        assertTrue(combatSystem.config.containsKey("partyIds"))
        assertTrue(combatSystem.config.containsKey("encounterData"))
    }
    @Test fun `combat scene uses battleUpdate producing TriggerSystem`() {
        val combatScene = ir.scenes.first { it.id == "combat_scene" }
        // battleUpdate("combat") should produce a TriggerSystem("combat") ScriptOp
        val hasTrigger = combatScene.frameOps.any { it is TriggerSystem && it.systemId == "combat" }
        assertTrue(hasTrigger, "Expected TriggerSystem('combat') in combat_scene frameOps")
    }
    @Test fun `gameplay scene has movement ops`() {
        val gameplay = ir.scenes.first { it.id == "gameplay" }
        assertTrue(gameplay.frameOps.isNotEmpty())
    }
    @Test fun `all IR types are core sealed types`() {
        // Verify the RPG builders produced only core IR types, no custom sealed subtypes
        fun assertCoreOps(ops: List<ScriptOp>) {
            ops.forEach { op ->
                // All ScriptOp subtypes must be from the core ir.v2 package
                assertTrue(op::class.qualifiedName?.startsWith("io.github.gbkt.core.ir.v2") == true,
                    "Non-core ScriptOp found: ${op::class.qualifiedName}")
                when (op) {
                    is IfOp -> { assertCoreOps(op.then); assertCoreOps(op.otherwise) }
                    is WhileOp -> assertCoreOps(op.body)
                    else -> {} // Leaf ops are fine
                }
            }
        }
        ir.scenes.forEach { scene ->
            assertCoreOps(scene.enterOps + scene.frameOps + scene.exitOps)
        }
    }
}
