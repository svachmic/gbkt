    private fun evaluateCallExpr(expr: CallExpr): Long {
        val fn = expr.function
        val args = expr.args.map { evaluateExpr(it) }

        return when {
            // Existing collision check
            fn == "collides" -> { ... }

            // Hash table operations: ht_{name}_{op}
            fn.startsWith("ht_") && fn.endsWith("_insert") -> {
                val name = fn.removePrefix("ht_").removeSuffix("_insert")
                hashTables.getOrPut(name) { mutableMapOf() }[args[0]] = args[1]
                0L
            }
            fn.startsWith("ht_") && fn.endsWith("_get") -> {
                val name = fn.removePrefix("ht_").removeSuffix("_get")
                hashTables[name]?.get(args[0]) ?: 0L
            }
            fn.startsWith("ht_") && fn.endsWith("_contains") -> {
                val name = fn.removePrefix("ht_").removeSuffix("_contains")
                if (hashTables[name]?.containsKey(args[0]) == true) 1L else 0L
            }
            fn.startsWith("ht_") && fn.endsWith("_remove") -> {
                val name = fn.removePrefix("ht_").removeSuffix("_remove")
                hashTables[name]?.remove(args[0])
                0L
            }
            fn.startsWith("ht_") && fn.endsWith("_size") -> {
                val name = fn.removePrefix("ht_").removeSuffix("_size")
                (hashTables[name]?.size ?: 0).toLong()
            }
            fn.startsWith("ht_") && fn.endsWith("_clear") -> {
                val name = fn.removePrefix("ht_").removeSuffix("_clear")
                hashTables[name]?.clear()
                0L
            }

            // Ring buffer operations: ring_{name}_{op}
            fn.startsWith("ring_") && fn.endsWith("_push") -> {
                val name = fn.removePrefix("ring_").removeSuffix("_push")
                ringBuffers.getOrPut(name) { ArrayDeque() }.addLast(args[0])
                0L
            }
            fn.startsWith("ring_") && fn.endsWith("_pop") -> {
                val name = fn.removePrefix("ring_").removeSuffix("_pop")
                ringBuffers[name]?.removeFirstOrNull() ?: 0L
            }
            fn.startsWith("ring_") && fn.endsWith("_peek") -> {
                val name = fn.removePrefix("ring_").removeSuffix("_peek")
                ringBuffers[name]?.firstOrNull() ?: 0L
            }
            fn.startsWith("ring_") && fn.endsWith("_size") -> {
                val name = fn.removePrefix("ring_").removeSuffix("_size")
                (ringBuffers[name]?.size ?: 0).toLong()
            }
            fn.startsWith("ring_") && fn.endsWith("_clear") -> {
                val name = fn.removePrefix("ring_").removeSuffix("_clear")
                ringBuffers[name]?.clear()
                0L
            }

            // Pool operations: pool_{name}_{op}
            fn.startsWith("pool_") && fn.endsWith("_alloc") -> {
                val name = fn.removePrefix("pool_").removeSuffix("_alloc")
                // Simple allocation: return next index
                val pool = hashTables.getOrPut("__pool_$name") { mutableMapOf() }
                val idx = pool.size.toLong()
                pool[idx] = 1L // 1 = active
                idx
            }
            fn.startsWith("pool_") && fn.endsWith("_free") -> {
                val name = fn.removePrefix("pool_").removeSuffix("_free")
                hashTables["__pool_$name"]?.remove(args[0])
                0L
            }
            fn.startsWith("pool_") && fn.endsWith("_active_count") -> {
                val name = fn.removePrefix("pool_").removeSuffix("_active_count")
                (hashTables["__pool_$name"]?.size ?: 0).toLong()
            }

            // Fixed slot operations: slot_{name}_{op}
            fn.startsWith("slot_") && fn.endsWith("_set") -> {
                val name = fn.removePrefix("slot_").removeSuffix("_set")
                fixedSlots.getOrPut(name) { mutableMapOf() }[args[0].toInt()] = args[1]
                0L
            }
            fn.startsWith("slot_") && fn.endsWith("_get") -> {
                val name = fn.removePrefix("slot_").removeSuffix("_get")
                fixedSlots[name]?.get(args[0].toInt()) ?: 0L

                val name = fn.removePrefix("slot_").removeSuffix("_clear")
                fixedSlots[name]?.remove(args[0].toInt())
                0L
            }

            else -> 0L // Unknown function — no-op stub
        }
    }
