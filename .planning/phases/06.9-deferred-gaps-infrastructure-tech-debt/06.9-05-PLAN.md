    **ScriptOp serialization:**
    ```kotlin
    private fun serializeOp(op: ScriptOp): JSONObject {
        val json = JSONObject()
        when (op) {
            is Assign -> {
                json.put("type", "Assign")
                json.put("target", op.target)
                json.put("value", serializeExpr(op.value))
            }
            is IfOp -> {
                json.put("type", "If")
                json.put("condition", serializeExpr(op.condition))
                json.put("thenBranch", JSONArray(op.thenBranch.map { serializeOp(it) }))
                json.put("elseBranch", JSONArray(op.elseBranch.map { serializeOp(it) }))
            }
            // ... all other subtypes
            else -> {
                // Unknown/genre-specific op
                json.put("type", "Unknown")
                json.put("class", op::class.qualifiedName ?: op::class.simpleName)
            }
        }
        return json
    }
    ```

    **ScriptOp deserialization:**
    ```kotlin
    private fun deserializeOp(json: JSONObject): ScriptOp {
        return when (val type = json.getString("type")) {
            "Assign" -> Assign(
                target = json.getString("target"),
                value = deserializeExpr(json.getJSONObject("value"))
            )
            "If" -> IfOp(
                condition = deserializeExpr(json.getJSONObject("condition")),
                thenBranch = deserializeOpList(json.getJSONArray("thenBranch")),
                elseBranch = deserializeOpList(json.optJSONArray("elseBranch")?.let { deserializeOpList(it) } ?: emptyList())
            )
            // ... all other subtypes
            else -> {
                System.err.println("WARNING: Unknown ScriptOp type '$type'; substituting RawOp placeholder")
                RawOp("// unsupported: $type")
            }
        }
    }
    ```
