    ```kotlin
    package io.github.gbkt.gradle.tasks

    import org.gradle.api.DefaultTask
    import org.gradle.api.file.RegularFileProperty
    import org.gradle.api.provider.Property
    import org.gradle.api.tasks.*
    import java.io.File

    abstract class EmulatorTestTask : DefaultTask() {
        @get:InputFile
        abstract val romFile: RegularFileProperty

        @get:Input @get:Optional
        abstract val sameboyTesterPath: Property<String>

        @get:Input @get:Optional
        abstract val mgbaPath: Property<String>

        @get:Input
        abstract val maxFrames: Property<Int>

        @get:Input
        abstract val timeoutSeconds: Property<Int>

        init {
            group = "verification"
            description = "Runs ROM in headless emulator for automated testing"
            maxFrames.convention(600)   // 10 seconds at 60fps
            timeoutSeconds.convention(30)
        }

        @TaskAction
        fun run() { ... }
    }
    ```

    4. `PATH` lookup

    **mGBA detection:** Reuse the pattern from existing `ValidateRomTask`.
