       ```kotlin
       private fun processUge(file: File, relPath: String, outDir: File) {
           val songName = file.nameWithoutExtension
           val outputC = File(outDir, "song_${songName}.c")
           val outputH = File(outDir, "song_${songName}.h")

           // Try to find uge2c tool
           val uge2cPath = findUge2c()
           if (uge2cPath == null) {
               logger.warn("WARNING: uge2c not found — cannot convert '${file.name}' to C. " +
                   "Install hUGEDriver (https://github.com/SuperDisk/hUGEDriver) and ensure uge2c is on PATH " +
                   "or set HUGEDRIVER_HOME environment variable.")
               return
           }

           // Shell out to uge2c
           val result = project.exec {
               commandLine(uge2cPath, file.absolutePath, "-o", outputC.absolutePath)
               isIgnoreExitValue = true
           }

           if (result.exitValue != 0) {
               logger.error("uge2c failed for '${file.name}' with exit code ${result.exitValue}")
           } else {
               logger.lifecycle("Converted ${file.name} → song_${songName}.c")
           }
       }
       ```

    4. Implement `findUge2c()` following the `GbdkToolchain` detection pattern. **Make `findUge2c()` internal (not private)** so it can be tested:
       ```kotlin
       internal fun findUge2c(): String? {
           // 1. HUGEDRIVER_HOME environment variable
           val envHome = System.getenv("HUGEDRIVER_HOME")
           if (envHome != null) {
               val tool = File(envHome, "uge2c")
               if (tool.exists() && tool.canExecute()) return tool.absolutePath
           }

           // 2. Check PATH
           val pathResult = try {
               ProcessBuilder("which", "uge2c").start().let {
                   val path = it.inputStream.bufferedReader().readText().trim()
                   it.waitFor()
                   if (it.exitValue() == 0 && path.isNotEmpty()) path else null
               }
           } catch (_: Exception) { null }
           if (pathResult != null) return pathResult

           // 3. Common installation paths
           val commonPaths = listOf(
               "/opt/hUGEDriver/uge2c",
               "${System.getProperty("user.home")}/hUGEDriver/uge2c",
               "/usr/local/bin/uge2c",
           )
           return commonPaths.firstOrNull { File(it).exists() && File(it).canExecute() }
       }
       ```
