       ```kotlin
       /** Builder for defining struct types. */
       class StructBuilder(private val name: String) {
           private val fields = mutableListOf<StructFieldDef>()

           /** Add a field with the given name and primitive type. */
           fun field(name: String, type: VarType) {
               require(fields.none { it.name == name }) { "Duplicate field name '$name' in struct '${this.name}'" }
               fields.add(StructFieldDef(name, type))
           }

           // Convenience type references
           val u8 = VarType.U8
           val u16 = VarType.U16
           val i8 = VarType.I8
           val i16 = VarType.I16

           fun build(): StructDef = StructDef(name, fields.toList())
       }

       /** Proxy for accessing fields of a struct variable via dot syntax. */
       class StructVar(val name: String, val structDef: StructDef) {
           /** Access a field by name. Returns an AssignableVar-like reference for the field. */
           operator fun get(fieldName: String): AssignableVar {
               requireNotNull(structDef.field(fieldName)) {
                   "Struct '${structDef.name}' has no field '$fieldName'. Available: ${structDef.fields.map { it.name }}"
               }
               return AssignableVar("${name}.${fieldName}")
           }
       }

       /** Declare a struct type. Registers in GameIR.structs. */
       fun GameBuilder.struct(name: String, block: StructBuilder.() -> Unit): StructDef {
           val builder = StructBuilder(name)
           builder.block()
           val structDef = builder.build()
           // Register struct in game builder (add a registerStruct method to GameBuilder)
           registerStruct(structDef)
           return structDef
       }
       ```
