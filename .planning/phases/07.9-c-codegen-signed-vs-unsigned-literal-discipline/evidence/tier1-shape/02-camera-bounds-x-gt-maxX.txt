INT16 rawX = (INT16)_player_x - 80u;
INT16 rawY = (INT16)_player_y - 72u;
_camera_x = (UINT8)((rawX < 0) ? 0u : (rawX > 96) ? 96u : rawX);
_camera_y = (UINT8)((rawY < 0) ? 0u : (rawY > 112) ? 112u : rawY);
if (_camera_shake_timer > 0u) {
    UINT8 offset = (_camera_shake_timer & 1u != 0u) ? _camera_shake_intensity : 0u;
    SCX_REG = _camera_x + offset;
    SCY_REG = _camera_y + offset;
    --_camera_shake_timer;
} else {
    SCX_REG = _camera_x;
    SCY_REG = _camera_y;
}
