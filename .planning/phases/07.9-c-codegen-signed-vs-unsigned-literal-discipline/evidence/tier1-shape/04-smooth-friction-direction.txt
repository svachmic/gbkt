if (dpad_held(J_LEFT)) {
    _player_vx -= ACCEL_PLAYER;
}
if (dpad_held(J_RIGHT)) {
    _player_vx += ACCEL_PLAYER;
}
if (!dpad_held(J_LEFT | J_RIGHT)) {
    if (_player_vx > 0) {
        _player_vx -= FRICTION_PLAYER;
        if (_player_vx < 0) {
            _player_vx = 0;
        }
    }
    if (_player_vx < 0) {
        _player_vx += FRICTION_PLAYER;
        if (_player_vx > 0) {
            _player_vx = 0;
        }
    }
}
if (_player_vx > SPEED_PLAYER) {
    _player_vx = SPEED_PLAYER;
}
if (_player_vx < -SPEED_PLAYER) {
    _player_vx = -SPEED_PLAYER;
}
if (dpad_held(J_UP)) {
    _player_vy -= ACCEL_PLAYER;
}
if (dpad_held(J_DOWN)) {
    _player_vy += ACCEL_PLAYER;
}
if (!dpad_held(J_UP | J_DOWN)) {
    if (_player_vy > 0) {
        _player_vy -= FRICTION_PLAYER;
        if (_player_vy < 0) {
            _player_vy = 0;
        }
    }
    if (_player_vy < 0) {
        _player_vy += FRICTION_PLAYER;
        if (_player_vy > 0) {
            _player_vy = 0;
        }
    }
}
if (_player_vy > SPEED_PLAYER) {
    _player_vy = SPEED_PLAYER;
}
if (_player_vy < -SPEED_PLAYER) {
    _player_vy = -SPEED_PLAYER;
}
_player_x += (UINT8)_player_vx;
_player_y += (UINT8)_player_vy;
