```kotlin
                    awardExp()

                onState(CombatState.DEFEAT) {
                    // Navigate to game over — wired in scene plan
                }

                // Damage formula (from stats.c)
                // Physical: (ATK * ability.power / DEF) with variance
                // Magic: (MATK * ability.power / MDEF) with variance

                // Status effect tick on turn start
                onTurnStart {
                    // DoT/HoT tick, duration decrement, expiry
                }
            }
        }
    }
}
```
