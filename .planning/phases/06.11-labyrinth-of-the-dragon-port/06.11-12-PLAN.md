    fun register(builder: GameBuilder) {
        builder.apply {
            scene("battle") {
                enter {
                    // Load battle tilemap
                    // Apply battle palettes
                    // Initialize party from selected class
                    // Initialize enemies from encounter data
                    // Display monster sprite
                    // Show battle UI (HP bars, menus)
                }
                every.frame {
                    // Drive combat state machine
                    battleUpdate(CombatSystem.combat)

                    // Handle combat states
                    whenever(combatIsInState(CombatState.PLAYER_TURN)) {
                        // Show action menu: Attack, Magic, Item, Flee
                        // D-pad navigates menu
                        // A selects action
                    }
                    whenever(combatIsInState(CombatState.TARGET_SELECT)) {
                        // Show target selection cursor
                        // D-pad moves between targets
                        // A confirms target
                    }
                    whenever(combatIsInState(CombatState.VICTORY)) {
                        // Show exp gained, loot drops
                        // Wait for input, then navigate back to gameplay
                        navigate(Scenes.gameplayRef)
                    }
                    whenever(combatIsInState(CombatState.DEFEAT)) {
                        navigate(Scenes.gameOverRef)
                    }
                }
                exit {
                    // Clean up battle sprites
                }
            }
        }
    }
}
