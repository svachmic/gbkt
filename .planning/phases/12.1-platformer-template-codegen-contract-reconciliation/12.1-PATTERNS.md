```kotlin
@GbktDsl
class TilemapCollisionBuilder(val id: String) {
    private var posXVar: String? = null
    private var posYVar: String? = null
    private var vxVar: String? = null
    private var vyVar: String? = null
    private var groundedVar: String? = null
    private var hitboxX: Int = 0
    private var hitboxY: Int = 0
    private var hitboxW: Int = 8
    private var hitboxH: Int = 24
    private var solidThreshold: Int = 17

    fun position(x: AssignableVar, y: AssignableVar) {
        posXVar = x.name; posYVar = y.name
    }
    fun velocity(vx: AssignableVar, vy: AssignableVar) {
        vxVar = vx.name; vyVar = vy.name
    }
    fun grounded(g: AssignableVar) { groundedVar = g.name }
    fun hitbox(x: Int, y: Int, w: Int, h: Int) {
        hitboxX = x; hitboxY = y; hitboxW = w; hitboxH = h
    }
    fun solidThreshold(v: Int) { solidThreshold = v }

    internal fun build(): GenericSystem = GenericSystem(
        id = id,
        config = mapOf(
            "type" to "tilemap_collision",
            "posXVar" to posXVar, "posYVar" to posYVar,
            "vxVar" to vxVar, "vyVar" to vyVar,
            "groundedVar" to groundedVar,
            "hitboxX" to hitboxX, "hitboxY" to hitboxY,
            "hitboxW" to hitboxW, "hitboxH" to hitboxH,
            "solidThreshold" to solidThreshold,
        ),
    )
}
```

**Import note:** `AssignableVar` lives in `io.github.gbkt.core.dsl.AssignableVar` (gbkt-lang). `@GbktDsl` lives in `io.github.gbkt.core.dsl.GbktDsl`. Both already imported by sibling builders in this module.

**Rationale:** A file-level `mainC.contains("_player_x")` would false-positive on the rect-physics path's emission. Brace-walk extracts the named function's body so substring checks fire only inside scope.
