override fun visitPrintOp(op: PrintOp): CStatement {
    val statements = mutableListOf<CStatement>()
    val position = op.position
    if (position != null) {
        statements +=
            CExprStatement(CCall("gotoxy", listOf(CLiteral(position.x), CLiteral(position.y))))

The architectural rule the RED tests enforce:
- A scene "has a BG tilemap" iff: `scene.tilesetRef != null` OR a genre system contributes enterOps for that scene (e.g., racing splice).
- For such scenes: the DSL `clear()` keyword MUST NOT result in a bare `cls()` in generated C (which wipes BG); the DSL `print()` keyword MUST NOT result in `gotoxy()/printf()` (which writes to BG layer).
- For scenes WITHOUT a BG tilemap (title, results, gameover): existing behavior is fine — user wants the BG cleared and HUD-style text printed.

        private fun extractFunctionBody(cSource: String, funcName: String): String {
            val signatureIdx = cSource.indexOf("void $funcName(void)")
            require(signatureIdx >= 0) { "$funcName not found in C source" }
            val openBrace = cSource.indexOf('{', signatureIdx)
            require(openBrace >= 0) { "open brace for $funcName not found" }

            var depth = 1
            var i = openBrace + 1
            while (i < cSource.length && depth > 0) {
                when (cSource[i]) { '{' -> depth++; '}' -> depth-- }
                i++
            }
            return cSource.substring(openBrace + 1, i - 1)
