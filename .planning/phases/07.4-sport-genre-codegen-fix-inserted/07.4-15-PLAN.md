// 1. Read target waypoint
val tgtX = wpX[wpIdx]
val tgtY = wpY[wpIdx]
// 2. Compute |dx|, |dy|
val dx = if (poolX > tgtX) poolX - tgtX else tgtX - poolX
val dy = if (poolY > tgtY) poolY - tgtY else tgtY - poolY
// 3. Cardinal heading from largest axis
val heading = if (dx >= dy) {
    if (poolX < tgtX) 1 /*E*/ else 3 /*W*/
} else {
    if (poolY < tgtY) 2 /*S*/ else 0 /*N*/
}
// 4. Throttle ramp toward effectiveCap (≈ aiSpeed * speedPercent / 100, with rubber-band).
//    For Test 2 the simplification `speedCur = min(speedCur + ((accel >> 4) + 1), effectiveCap)`
//    is sufficient — Plan 05's RacingMovementMathTest exercises the exact ramp shape.
val accelStep = ((aiAccel shr 4) + 1)
speedCur = if (speedCur + accelStep < effectiveCap) speedCur + accelStep else effectiveCap
// 5. delta = speed_cur >> 5
val delta = speedCur shr 5
val (vx, vy) = when (heading) {
    0 -> 0 to -delta
    1 -> delta to 0
    2 -> 0 to delta
    3 -> -delta to 0
    else -> 0 to 0
}
// 6. Wall-collision write-back (Plan 12 logic)
val propXs = poolX + vx  // INT16 promotion
val propYs = poolY + vy
val maxX = mapWidth * 8 - 8   // sprite_w = 8
val maxY = mapHeight * 8 - 16 // sprite_h = 16
if (propXs in 0 until maxX && propYs in 0 until maxY) {
    val sampleX = propXs + 4   // sprite half-W
    val sampleY = propYs + 8   // sprite half-H
    val tileCol = sampleX shr 3
    val tileRow = sampleY shr 3
    if (tileCol < mapWidth && tileRow < mapHeight) {
        val tile = tiles[tileRow * mapWidth + tileCol]
        if (tile != 0) {
            poolX = propXs
            poolY = propYs
        }
    }
}
// 7. Waypoint advance — when within proximity threshold, advance wp_idx
val proximityPx = 12 // approximate from SportVisitor — verify by reading buildAiPoolBodyStatements
if (dx <= proximityPx && dy <= proximityPx) {
    wpIdx = (wpIdx + 1) % waypointCount
}

    private data class AiTranscript(
        val finalX: Int,
        val finalY: Int,
        val finalWpIdx: Int,
        val lapsCompleted: Int,
        val frozenAtX: Int,
        val frozenAtY: Int,
        val maxFrozenStreak: Int,
    )

    @Suppress("LongParameterList", "ComplexMethod")
    private fun simulateAi(
        tiles: List<Int>,
        mapW: Int,
        mapH: Int,
        wpX: List<Int>,
        wpY: List<Int>,
        startPx: Int,
        startPy: Int,
        speedCap: Int,
        accel: Int,
        frameBudget: Int,
        lapsTarget: Int,
    ): AiTranscript {
        var px = startPx
        var py = startPy
        var wpIdx = 0
        var speedCur = 0
        var lapsCompleted = 0
        var prevPx = px
        var prevPy = py
        var frozenStreak = 0
        var maxFrozenStreak = 0
        var frozenAtX = px
        var frozenAtY = py
        val accelStep = ((accel shr 4) + 1)

        for (frame in 0 until frameBudget) {
            val tgtX = wpX[wpIdx]
            val tgtY = wpY[wpIdx]
            val dx = if (px > tgtX) px - tgtX else tgtX - px
            val dy = if (py > tgtY) py - tgtY else tgtY - py

            val heading = if (dx >= dy) {
                if (px < tgtX) 1 else 3
            } else {
                if (py < tgtY) 2 else 0
            }

            speedCur = if (speedCur + accelStep < speedCap) speedCur + accelStep else speedCap
            val delta = speedCur shr 5
            val vx = when (heading) { 1 -> delta; 3 -> -delta; else -> 0 }
            val vy = when (heading) { 0 -> -delta; 2 -> delta; else -> 0 }

            val propXs = px + vx
            val propYs = py + vy
            val maxX = mapW * 8 - 8
            val maxY = mapH * 8 - 16
            if (propXs in 0 until maxX && propYs in 0 until maxY) {
                val sampleX = propXs + 4
                val sampleY = propYs + 8
                val tileCol = sampleX shr 3
                val tileRow = sampleY shr 3
                if (tileCol < mapW && tileRow < mapH) {
                    val tile = tiles[tileRow * mapW + tileCol]
                    if (tile != TILE_WALL) {
                        px = propXs
                        py = propYs
                    }
                }
            }

            // Waypoint advance
            if (dx <= AI_PROXIMITY_PX && dy <= AI_PROXIMITY_PX) {
                wpIdx = (wpIdx + 1) % wpX.size
                if (wpIdx == 0) lapsCompleted += 1
                if (lapsCompleted >= lapsTarget) {
                    return AiTranscript(
                        finalX = px, finalY = py, finalWpIdx = wpIdx,
                        lapsCompleted = lapsCompleted,
                        frozenAtX = frozenAtX, frozenAtY = frozenAtY,
                        maxFrozenStreak = maxFrozenStreak,
                    )
                }
            }

            if (px == prevPx && py == prevPy) {
                frozenStreak += 1
                if (frozenStreak > maxFrozenStreak) {
                    maxFrozenStreak = frozenStreak
                    frozenAtX = px
                    frozenAtY = py
                }
            } else {
                frozenStreak = 0
            }
            prevPx = px
            prevPy = py
        }

        return AiTranscript(
            finalX = px, finalY = py, finalWpIdx = wpIdx,
            lapsCompleted = lapsCompleted,
            frozenAtX = frozenAtX, frozenAtY = frozenAtY,
            maxFrozenStreak = maxFrozenStreak,
        )
    }

    private fun debugTileNeighborhood(
        tiles: List<Int>,
        mapW: Int,
        cx: Int,
        cy: Int,
    ): String {
        val sb = StringBuilder()
        for (dy in -2..2) {
            for (dx in -2..2) {
                val nx = cx + dx
                val ny = cy + dy
                if (nx < 0 || ny < 0 || nx >= mapW || ny * mapW + nx >= tiles.size) {
                    sb.append("? ")
                } else {
                    sb.append("${tiles[ny * mapW + nx]} ")
                }
            }
            sb.append("| ")
        }
        return sb.toString()
    }
}
