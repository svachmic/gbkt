class RacingValidationPass : AnalysisPass {
    override fun run(context: PassContext): PassResult {
        val game = context.game
        val diagnostics = mutableListOf<Diagnostic>()

        val racingSystems = game.systems.filterIsInstance<GenericSystem>()
            .filter { it.config["type"] == "sport_racing" }

        for (sys in racingSystems) {
            val racingConfig = sys.config["config"] as? RacingConfig ?: continue
            val racingId = racingConfig.id

            // Check 1: ANLZ-RACING-01 — polygon non-degeneracy.
            checkPolygonNonDegenerate(racingConfig, racingId, diagnostics)

            // Check 2: ANLZ-RACING-02 — racing has a player.
            checkPlayerBound(racingConfig, racingId, diagnostics)

            // Check 3: ANLZ-RACING-03 — every referenced vehicle has actor binding.
            checkVehicleActorBindings(game, racingConfig, racingId, diagnostics)

            // Check 4: ANLZ-RACING-04 — scene frameOps don't contain moveBy(bound player) or direct .x/.y writes.
            checkNoHandCodedMovement(game, racingConfig, diagnostics)

            // Check 5: ANLZ-RACING-05 — user CameraSystem follows the bound player actor.
            checkCameraFollowsPlayer(game, racingConfig, diagnostics)
        }

        val errors = diagnostics.filter { it.severity == Severity.ERROR }
        return if (errors.isNotEmpty()) PassResult.Failed(diagnostics)
               else PassResult.Success(context.withDiagnostics(diagnostics))
    }

    // ... private helpers ...
}
