```kotlin
/**
 * Build the per-race-scene enter ops that close GAP-A (AI pool activation), GAP-B (track
 * tileset + tilemap load), and GAP-D (camera target bind). Ops are emitted in this order so
 * that the runtime VRAM is correct before any sprites become visible.
 */
private fun buildRaceEnterOps(
    racingId: String,
    playerActorId: String,
    resolvedAiVehicles: List<Vehicle>,
    gameIR: GameIR,
): List<ScriptOp> {
    val ops = mutableListOf<ScriptOp>()

    // GAP-D: camera target bind
    val playerActorIdx = gameIR.actors.indexOfFirst { it.id == playerActorId }
    if (playerActorIdx >= 0) {
        ops += RawOp("_camera_target = ${playerActorIdx}u;")
    }

    // GAP-B: tileset + tilemap load
    val trackZone = gameIR.zones.firstOrNull { it.id == racingId }
    if (trackZone != null) {
        ops += RawOp("set_bkg_data(0, 3, _racing_${racingId}_tileset);")
        ops += RawOp(
            "set_bkg_tiles(0, 0, ${trackZone.mapWidth}u, ${trackZone.mapHeight}u, _zone_${racingId}_tiles);"
        )
        // Value `1u` matches the existing dungeon path's tileset-loaded sentinel (see
        // GBDKPipelineV2.addTilesetGuardToEnterFunction). If addTilesetGuardToEnterFunction
        // already injects this sentinel write for the racing path, this RawOp is redundant —
        // verify during implementation and either delete or keep with this comment in place.
        ops += RawOp("_current_tileset_id = 1u;")
    }

    // GAP-A: AI pool spawn (one call per AI vehicle slot, position from the bound rival actor)
    for (aiVehicle in resolvedAiVehicles) {
        val aiVehicleId = sanitizeCId(aiVehicle.id)
        val rivalActor = gameIR.actors.firstOrNull { it.id == aiVehicle.actorRef.id }
        val pos = rivalActor?.position
        if (pos != null) {
            ops += RawOp("pool_${aiVehicleId}_spawn(${pos.x}u, ${pos.y}u);")
        }
    }

    return ops
}

/**
 * Builtin 3-tile tileset emitted as a 48-byte const array. Tiles 0, 1, 2 match the
 * TrackSynthesizer constants (Plan 07.4-04): 0=wall (filled), 1=drivable (open),
 * 2=grass (checkerboard). Each Game Boy tile is 16 bytes (8x8 px, 2 bpp interleaved).
 *
 * Hand-tuned graphics — sufficient for runtime visibility (track corridor renders as
 * distinct from grass / wall). Future phases may expose a `tileset(asset(...))` slot on
 * racing { } to override; for 07.4 the builtin is the only source.
 */
private fun buildBuiltinTrackTilesetVarDecl(racingId: String): CVarDecl {
    val bytes = builtinTrackTilesetBytes()
    require(bytes.size == 48) { "Builtin track tileset must be 48 bytes (3 tiles * 16); got ${bytes.size}" }
    return CVarDecl(
        name = "_racing_${racingId}_tileset",
        type = CArray(CConst(CU8), 48),
        initializer = CRawExpr("{ ${bytes.joinToString(", ") { "0x%02X".format(it) }} }"),
    )
}

private fun builtinTrackTilesetBytes(): List<Int> = listOf(
    // Tile 0 — WALL (solid filled, all 16 bytes 0xFF). 8x8 black square.
    0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF,
    0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF,
    // Tile 1 — DRIVABLE (open, all 16 bytes 0x00). 8x8 white/light square.
    0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00,
    0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00,
    // Tile 2 — GRASS (checkerboard pattern; alternating 0x55 / 0xAA per row gives a
    // visually distinct dotted texture from the open road).
    0x55, 0x00, 0xAA, 0x00, 0x55, 0x00, 0xAA, 0x00,
    0x55, 0x00, 0xAA, 0x00, 0x55, 0x00, 0xAA, 0x00,
)
```
