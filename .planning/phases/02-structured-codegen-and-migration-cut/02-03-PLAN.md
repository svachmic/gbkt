```kotlin
package io.github.gbkt.backend.gbdk.codegen.visitor

object ExprVisitor {
    fun visit(expr: Expr): CExpr = when (expr) {
        is Literal -> CLiteral(expr.value)
        is StringLiteral -> CStringLiteral(expr.value)
        is VarRef -> CVar(sanitizeVarName(expr.name))
        is BinaryExpr -> CBinaryExpr(visit(expr.left), binaryOpToC(expr.op), visit(expr.right))
        is UnaryExpr -> CUnaryExpr(unaryOpToC(expr.op), visit(expr.operand))
        is CallExpr -> CCall(expr.name, expr.args.map { visit(it) })
        is TernaryExpr -> CTernary(visit(expr.condition), visit(expr.thenExpr), visit(expr.elseExpr))
        is ArrayAccessExpr -> CArrayAccess(visit(expr.array), visit(expr.index))
        is PropertyAccessExpr -> CVar(sanitizeVarName("${/* base name */}_${expr.property}"))
    }
}
```

Variable name sanitization: replace dots with underscores, prepend underscore for GBDK naming convention (e.g., `ball.x` → `_ball_x`). Note: check if variables from PongV2 use dot notation (they do — `varRef("ball.x")`).

BinaryOp mapping: ADD→"+", SUB→"-", MUL→"*", DIV→"/", MOD→"%", AND→"&", OR→"|", XOR→"^", SHL→"<<", SHR→">>", EQ→"==", NEQ→"!=", LT→"<", LTE→"<=", GT→">", GTE→">=", LOGICAL_AND→"&&", LOGICAL_OR→"||"

**SceneVisitor.kt:**
```kotlin
object SceneVisitor {
    fun visit(scene: SceneIR): List<CFunction> {
        val functions = mutableListOf<CFunction>()
        if (scene.enterOps.isNotEmpty()) {
            functions += CFunction(
                name = "${scene.id}_enter",
                returnType = CVoid,
                body = scene.enterOps.map { ScriptOpVisitor.visit(it) },
                isBanked = true,
                sectionComment = "Scene: ${scene.id}",
            )
        }
        // Same for frameOps (${scene.id}_frame) and exitOps (${scene.id}_exit)
        // frameOps and exitOps don't repeat the sectionComment
        return functions
    }

    fun generateSceneEnum(sceneIds: List<String>): List<CDefine> {
        return sceneIds.mapIndexed { index, id ->
            CDefine("SCENE_${id.uppercase()}", "$index")
        }
    }
}
```

**ActorVisitor.kt:**
```kotlin
object ActorVisitor {
    fun visit(actor: ActorIR): List<CVarDecl> {
        val decls = mutableListOf<CVarDecl>()
        val prefix = "_${sanitize(actor.id)}"
        decls += CVarDecl("${prefix}_x", CU8, CLiteral(actor.position.x))
        decls += CVarDecl("${prefix}_y", CU8, CLiteral(actor.position.y))
        // Sprite data handled as comments or raw code (asset pipeline is Phase 3)
        return decls
    }
}
```
