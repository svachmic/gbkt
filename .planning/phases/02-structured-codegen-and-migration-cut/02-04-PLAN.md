```kotlin
package io.github.gbkt.backend.gbdk.codegen.pipeline

class GBDKPipelineV2 {
    fun generate(gameIR: GameIR): Map<String, String> {
        val cFiles = buildCFiles(gameIR)
        return cFiles.associate { cFile ->
            cFile.name to CEmitter.emit(cFile)
        }
    }

    private fun buildCFiles(gameIR: GameIR): List<CFile> {
        // 1. Build main.c (bank 0 — home file)
        val homeFile = buildHomeFile(gameIR)
        // 2. Build bank1.c (bank 1 — scene functions)
        val sceneFile = buildSceneFile(gameIR)
        return listOf(homeFile, sceneFile)
    }
}
```

**Home file (main.c, bank 0) contains:**
- `#include <gb/gb.h>` and `#include "game.h"` (standard GBDK includes)
- Scene enum defines: `#define SCENE_TITLE 0`, `#define SCENE_GAME 1`, `#define SCENE_GAMEOVER 2` (from SceneVisitor.generateSceneEnum)

- Global variable declarations: `UINT8 p1Score = 0u;` etc. (from GameIR.variables, mapping VarType to CType)

- `main()` function with:
  - Variable initialization

```kotlin
/**
 * Generate C source files from a v2 GameIR using the new typed C AST pipeline.
 * This is the replacement for the string-based GBDKCodeGenerator.
 */
fun generateV2(gameIR: GameIR): GenerationResult {
    return try {
        val pipeline = GBDKPipelineV2()
        val files = pipeline.generate(gameIR)
        val generatedFiles = files.mapValues { (path, content) ->
            GeneratedFile(path, content, "Generated by GBDKPipelineV2")
        }
        GenerationResult(success = true, files = generatedFiles)
    } catch (e: Exception) {
        GenerationResult.failed(e.message ?: "V2 code generation failed")
    }
}
```
