<!-- Existing gate (GenerateCTask.kt lines 747-813) — reflection pattern to replicate -->
@Suppress("UNCHECKED_CAST")
internal fun validateMetaspriteSpritePaths(gameIR: Any) {
    val metasprites = try {
        gameIR.javaClass.getMethod("getMetasprites").invoke(gameIR) as List<Any>
    } catch (_: NoSuchMethodException) { return }

    for (ms in metasprites) {
        val spritePath = try {
            ms.javaClass.getMethod("getSpritePath").invoke(ms) as String?
        } catch (_: NoSuchMethodException) { continue }

        if (spritePath == null) {
            val id = try { ms.javaClass.getMethod("getId").invoke(ms) as String }
                     catch (_: Exception) { "<unknown>" }
            throw GradleException(
                "Metasprite '$id' is missing sprite(asset(...)) ..."
            )
        }
    }
}
