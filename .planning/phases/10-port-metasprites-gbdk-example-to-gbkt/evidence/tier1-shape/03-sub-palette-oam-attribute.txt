void play_frame(void) BANKED {
    uint8_t hiwater = 0u;
    if (dpad_held(J_UP)) {
        _spdY -= 2u;
        if (_spdY < -32) {
            _spdY = -32;
        }
    }
    if (dpad_held(J_DOWN)) {
        _spdY += 2u;
        if (_spdY > 32) {
            _spdY = 32u;
        }
    }
    if (dpad_held(J_LEFT)) {
        _spdX -= 2u;
        if (_spdX < -32) {
            _spdX = -32;
        }
    }
    if (dpad_held(J_RIGHT)) {
        _spdX += 2u;
        if (_spdX > 32) {
            _spdX = 32u;
        }
    }
    if (button_pressed(J_B)) {
        _idx = _idx + 1u;
        if (_idx >= 5u) {
            _idx = 0u;
        }
    }
    if (button_pressed(J_A)) {
        _rot = _rot + 1u;
        _rot = _rot & 15u;
    }
    _posX += _spdX;
    _posY += _spdY;
    {
    uint8_t subpal = _rot >> 2;
    _elephant_subPalette = subpal;
    _elephant_flipX = (_rot & 0x3u) >> 0u;
    _elephant_flipY = (_rot & 0x3u) >> 1u;
    switch (_rot & 0x3u) {
        case 1:
            hiwater += move_metasprite_flipy(elephant_metasprites[_idx], 0, subpal, hiwater,
                                              DEVICE_SPRITE_PX_OFFSET_X + (_posX >> 4),
                                              DEVICE_SPRITE_PX_OFFSET_Y + (_posY >> 4));
            break;
        case 2:
            hiwater += move_metasprite_flipxy(elephant_metasprites[_idx], 0, subpal, hiwater,
                                              DEVICE_SPRITE_PX_OFFSET_X + (_posX >> 4),
                                              DEVICE_SPRITE_PX_OFFSET_Y + (_posY >> 4));
            break;
        case 3:
            hiwater += move_metasprite_flipx(elephant_metasprites[_idx], 0, subpal, hiwater,
                                              DEVICE_SPRITE_PX_OFFSET_X + (_posX >> 4),
                                              DEVICE_SPRITE_PX_OFFSET_Y + (_posY >> 4));
            break;
        default:
            hiwater += move_metasprite_ex(elephant_metasprites[_idx], 0, subpal, hiwater,
                                          DEVICE_SPRITE_PX_OFFSET_X + (_posX >> 4),
                                          DEVICE_SPRITE_PX_OFFSET_Y + (_posY >> 4));
            break;
    }
}

    if (_spdY < 0) {
        _spdY += 1u;
    }
    if (_spdY > 0) {
        _spdY -= 1u;
    }
    if (_spdX < 0) {
        _spdX += 1u;
    }
    if (_spdX > 0) {
        _spdX -= 1u;
    }
    hide_sprites_range(hiwater, MAX_HARDWARE_SPRITES);
}
