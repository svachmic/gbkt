    **GameDiscovery.kt** (`io.github.gbkt.test`):
    ```kotlin
    object GameDiscovery {
        fun configForGame(gameName: String, screenshotDir: File? = null): AgentSessionConfig? {
            val romFile = File("build/gbkt/output/$gameName.gb")
            if (!romFile.exists()) return null
            val resolvedScreenshotDir = screenshotDir ?: File("build/gbkt/test-failures")
            return AgentSessionConfig.discoverFiles(romFile, resolvedScreenshotDir)
        }

        data class GameInfo(val name: String, val romFile: File, val hasMetadata: Boolean)

        fun scanForBuiltRoms(projectRoot: File = File(".")): List<GameInfo> {
            val results = mutableListOf<GameInfo>()
            // Standalone: build/gbkt/output/*.gb
            val standaloneDir = File(projectRoot, "build/gbkt/output")
            if (standaloneDir.isDirectory) {
                standaloneDir.listFiles { f -> f.extension == "gb" }?.forEach { rom ->
                    val metadataFile = File(projectRoot, "build/gbkt/generated/game_metadata.json")
                    results.add(GameInfo(rom.nameWithoutExtension, rom, metadataFile.exists()))
                }
            }
            // Multi-game: gbkt-examples/*/build/gbkt/output/*.gb
            val examplesDir = File(projectRoot, "gbkt-examples")
            if (examplesDir.isDirectory) {
                examplesDir.listFiles { f -> f.isDirectory }?.forEach { gameDir ->
                    val outputDir = File(gameDir, "build/gbkt/output")
                    outputDir.listFiles { f -> f.extension == "gb" }?.forEach { rom ->
                        val metadataFile = File(gameDir, "build/gbkt/generated/game_metadata.json")
                        results.add(GameInfo(rom.nameWithoutExtension, rom, metadataFile.exists()))
                    }
                }
            }
            return results.sortedBy { it.name }
        }
    }
    ```

    Implements `BeforeEachCallback`, `AfterEachCallback`, `TestExecutionExceptionHandler`.
    - Constructor: `GbktTestExtension(gameName: String, customRomFile: File? = null, bootScript: ((StepAgent) -> Unit)? = null)`
    - Property: `val agent: StepAgent` (lateinit, exposed for test access)
    - Property: `val metadata: GameMetadata?` (nullable, loaded from config)
    - `beforeEach()`:
      - If `customRomFile != null`, use `AgentSessionConfig.discoverFiles(customRomFile, screenshotDir)`. Else use `GameDiscovery.configForGame(gameName)`.
      - If config is null, call `org.junit.jupiter.api.Assumptions.assumeTrue(false, "ROM not found for game '$gameName' — run buildRom first")`.
      - Load `GameMetadata` from `config.metadataFile` if it exists.
      - Create `StepAgent(config, metadata)` and call `agent.start()`.
    - `afterEach()`: If agent is initialized, call `agent.close()`.
    - `handleTestExecutionException()`: If agent is initialized, capture screenshot with label `"failure_{TestClass}_{testName}_frame{agent.lastFrame}"`. Also dump variables to a JSON sidecar file at `build/gbkt/test-failures/{TestClass}_{testName}_frame{N}.json` containing `{frame, scene, variables}`. Then re-throw.
    - Convenience delegation methods: `step(...)`, `stepN(...)`, `readVariable(...)`, `writeVariable(...)`, `waitForScene(...)`, `captureScreenshot(...)`.
