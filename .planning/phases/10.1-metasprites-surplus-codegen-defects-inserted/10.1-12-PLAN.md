DSL skeleton per RESEARCH §Synthetic ROM lines 827-875:
```kotlin
val metaspritesStress = game("MetaspritesStress") {
    config {
        cartridge = "ROM_ONLY"
        romBanks = 4
        target(GbcTarget.GBC_COMPATIBLE)
    }
    // Per-metasprite scoped variables (D-10 implicit scope-walk shape)
    var elephantPosX by i16Var(1280); var elephantPosY by i16Var(1152)
    var elephantIdx  by u8Var(0);     var elephantRot  by u8Var(0)
    var tigerPosX    by i16Var(640);  var tigerPosY    by i16Var(720)
    var tigerIdx     by u8Var(0);     var tigerRot     by u8Var(0)

    val player by actor {
        position(40, 40)
        sprite(asset("sprites/player.png")) { size(8, 16); hitbox(0, 0, 8, 16) }
    }
    val elephant by metasprite {
        posX(elephantPosX); posY(elephantPosY); idx(elephantIdx); rot(elephantRot)
        frame { tile(0, 0, 0) }
    }
    val tiger by metasprite {
        posX(tigerPosX); posY(tigerPosY); idx(tigerIdx); rot(tigerRot)
        frame { tile(0, 0, 0) }
    }
    val playScene = scene("play") {
        enter { showSprites() }
        frame {
            moveMetasprite(elephant)
            moveMetasprite(tiger)
        }
    }
    val titleScene = scene("title") {
        enter { showSprites() }
        frame { whenever(buttons.start.pressed) { navigate(playScene) } }
    }
    start = titleScene.id
}
```
