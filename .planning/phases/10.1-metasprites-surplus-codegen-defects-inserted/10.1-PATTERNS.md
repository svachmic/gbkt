```kotlin
fun palette(palette: GBCPalette) {
    // When slot == -1 (auto-assign), use the current paletteOps count as the next sequential
    // slot index. This ensures that palette(gray); palette(pink); palette(cyan); palette(green)
    // emits set_sprite_palette(0u,…), set_sprite_palette(1u,…), etc. — not all slot 0.

    paletteOps += SetPalette(palette.name, slot, palette.type)
}
```

actor index, not the auto-slot count — RESEARCH §Pattern 6 explicitly warns against the seed's
`mapIndexedNotNull` proposal).

**Fixed shape (RESEARCH §Pattern 6 line 634-643):**

```kotlin
package io.github.gbkt.backend.gbdk.codegen.pipeline

/**
 * Per-build singleton that hands out monotonically-increasing sprite-VRAM tile start indices.
 *
 * Replaces two parallel `var nextTile = 0` counters that previously lived in
 * [GBDKPipelineV2.buildSpriteDataLoadStatements] and [GBDKPipelineV2.buildMetaspriteTileDataLoadStatements].
 * Those counters provably collided when both actors and metasprites coexisted in the same game
 * (SEED-008 / D-08 / CR-01). One allocator instance per [GBDKPipelineV2.generate] call is monotonic
 * by construction.
 *
 * **Scope:** Phase 10.1 ships the per-build singleton variant. Per-scene / per-bank allocation
 * variants belong to Phase 11 (banks) when banked sprite tile loading becomes meaningful.
 * Keep the API surface minimal — no preemptive Phase 11 design (D-08 Claude's discretion).
 *
 * **Not exposed to user DSL** (D-13b) — internal codegen class only.
 */
internal class VramAllocator {
    private var nextTile: Int = 0

    /** Reserve [tileCount] contiguous sprite-VRAM slots. Returns the start index. */
    fun reserve(tileCount: Int): Int {
        require(tileCount >= 0) { "tileCount must be >= 0, got $tileCount" }
        val start = nextTile
        nextTile += tileCount
        return start
    }

    /** Total tiles reserved so far — useful for tests/diagnostics. */
    val tilesUsed: Int get() = nextTile
}
```

```kotlin
private fun extractFunctionBody(cSource: String, functionName: String): String {
    val lines = cSource.lines()
    val startIdx = lines.indexOfFirst { it.contains("void $functionName(") }
    if (startIdx == -1) return ""
    val body = StringBuilder()
    var depth = 0
    var started = false
    for (i in startIdx until lines.size) {
        val line = lines[i]
        body.appendLine(line)
        for (ch in line) {
            if (ch == '{') { depth++; started = true }
            if (ch == '}') depth--
        }
        if (started && depth == 0) break
    }
    return body.toString()
}
```
