```kotlin
private fun wrapFrameWithMetaspriteHiwater(
    functions: List<CFunction>,
    scene: SceneIR,
): List<CFunction> {
    val hasMetaspriteOps = scene.frameOps.any { it is io.github.gbkt.core.ir.MoveMetasprite }
    if (!hasMetaspriteOps) return functions
    return functions.map { fn ->
        if (fn.name != "${scene.id}_frame") return@map fn
        val prelude = listOf<CStatement>(CRawStatement("uint8_t hiwater = 0u;"))
        val postlude = listOf<CStatement>(
            CExprStatement(CCall("hide_sprites_range",
                listOf(CVar("hiwater"), CVar("MAX_HARDWARE_SPRITES"))))
        )
        fn.copy(body = prelude + fn.body + postlude)
    }
}
```
