       ```kotlin
       data class TacticalGridConfig(
           val gridWidth: Int = 8,           // grid columns
           val gridHeight: Int = 8,          // grid rows
           val enableTerrain: Boolean = true, // terrain cost effects
           val enableElevation: Boolean = false,
           val enableFacing: Boolean = false,  // flanking/backstab bonuses
           val flankingBonus: Int = 25,       // +25% damage from side
           val backstabBonus: Int = 50,       // +50% damage from behind
           val terrainTypes: List<TerrainTypeDef> = defaultTerrainTypes(),
           val elevationDamageBonus: Int = 10, // +10% per height advantage
       )

       data class TerrainTypeDef(
           val id: String,
           val name: String,
           val movementCost: Int = 1,        // 1=normal, 2=slow, 0=blocked, -1=impassable
           val damagePerTurn: Int = 0,        // terrain DOT
           val defenseBonus: Int = 0,         // cover
       )

       enum class FacingDirection { NORTH, SOUTH, EAST, WEST }
       ```

       - `fun enableFacing(flanking: Int = 25, backstab: Int = 50)`
       - `fun terrain(id: String, block: TerrainTypeBuilder.() -> Unit)`

       - Emit grid state: `_tgrid_<id>_terrain[W*H]` terrain type array, `_tgrid_<id>_elevation[W*H]` height map
