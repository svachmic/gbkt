```kotlin
// =============================================================================
// ARRAY PROPERTY DELEGATE
// =============================================================================

/**
 * Property delegate for declaring a global array variable.
 *
 * Registers an [ArrayDef] with the current [GameBuilder] via [GameBuilderContext]. Returns the
 * array name as a plain String so it can be used directly with [ScriptBuilder.arrayRef] and
 * [ScriptBuilder.arrayAssign].
 *
 * Usage:
 * ```kotlin
 * game("Breakout") {
 *     val bricks by u8Array("bricks", 30)
 *     // bricks = "bricks" — use in script ops as arrayRef("bricks", ...)
 * }
 * ```
 */
class ArrayDelegate(
    private val arrayName: String,
    private val elementType: VarType,
    private val size: Int,
) : ReadWriteProperty<Any?, String> {
    operator fun provideDelegate(
        thisRef: Any?,
        property: KProperty<*>,
    ): ReadWriteProperty<Any?, String> {
        GameBuilderContext.current?.registerArray(ArrayDef(arrayName, elementType, size))
        return this
    }

    override fun getValue(thisRef: Any?, property: KProperty<*>): String = arrayName

    override fun setValue(thisRef: Any?, property: KProperty<*>, value: String) {
        // No-op: array identity is set at declaration time
    }
}

/** Declares a global UINT8 array with [name] and [size] elements. Returns the array name. */
fun u8Array(name: String, size: Int) = ArrayDelegate(name, VarType.U8, size)
```
