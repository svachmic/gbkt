    ```kotlin
    object BudgetReporter {
        fun formatReport(context: PassContext): String = buildString {
            val game = context.game
            val config = context.config
            // Header
            appendLine("=" .repeat(56))
            appendLine(" gbkt Budget Report${" ".repeat(max(0, 36 - game.name.length))}${game.name}")
            appendLine("=".repeat(56))
            appendLine()

            // ROM Banks section
            formatBankSection(this, context)
            appendLine()

            // VRAM Tile Budget (per scene) table
            formatVRAMTable(this, context)
            appendLine()

            // Summary lines
            val oamCount = context.oamAssignments.size
            appendLine(" OAM Sprites: $oamCount / 40 (${oamCount * 100 / 40}%)")
            val ram = context.ramLayout
            appendLine(" WRAM: ${ram?.wramUsed ?: 0} / ${context.profile.memory.workRam} bytes (${...}%)")
            appendLine(" HRAM: ${ram?.hramUsed ?: 0} / 127 bytes (${...}%)")
            appendLine()

            // Error/warning count
            val errors = context.diagnostics.count { it.severity == Severity.ERROR }
            val warnings = context.diagnostics.count { it.severity == Severity.WARNING }
            appendLine(" $errors errors, $warnings warnings")
            appendLine("=".repeat(56))
        }
    }
    ```

    ```kotlin
    fun generateV2(gameIR: GameIR): GenerationResult {
        return try {
            // 1. Run analysis pipeline
            val analysisConfig = AnalysisConfig.fromCartridgeConfig(gameIR.config)
            val pipeline = DefaultPipeline.create()
            val initialContext = PassContext(
                game = gameIR,
                profile = profile,
                config = analysisConfig,
            )
            val analysisResult = pipeline.execute(initialContext)

            // Print budget report if available (shown during every build per locked decision)
            if (analysisResult is PassResult.Success) {
                analysisResult.context.budgetReport?.let { println(it) }
            }

            if (analysisResult is PassResult.Failed) {
                val errorMessages = analysisResult.diagnostics
                    .filter { it.severity == Severity.ERROR }
                    .joinToString("\n") { "${it.id}: ${it.message}" }
                return GenerationResult.failed("Analysis failed:\n$errorMessages")
            }

            val annotatedContext = (analysisResult as PassResult.Success).context

            // 2. Apply bank annotations to GameIR (copy scenes/actors with filled platform fields)
            val annotatedGame = applyAnnotations(annotatedContext.game, annotatedContext)

            // 3. Generate C code from annotated IR
            val pipelineV2 = GBDKPipelineV2()
            val files = pipelineV2.generate(annotatedGame)
            val generatedFiles = files.mapValues { (path, content) ->
                GeneratedFile(path, content, "Generated by GBDKPipelineV2")
            }
            GenerationResult(success = true, files = generatedFiles)
        } catch (e: Exception) {
            GenerationResult.failed(e.message ?: "V2 code generation failed")
        }
    }
    ```

      ```kotlin
      private fun applyAnnotations(game: GameIR, context: PassContext): GameIR {
          val annotatedScenes = game.scenes.map { scene ->
              scene.copy(
                  bankSlot = context.bankAssignments[scene.id],
                  vramRange = context.vramAssignments[scene.id],
              )
          }
          val annotatedActors = game.actors.map { actor ->
              actor.copy(
                  bankSlot = context.bankAssignments[actor.id],
                  vramRange = context.vramAssignments[actor.id],
                  oamSlot = context.oamAssignments[actor.id],
              )
          }
          return game.copy(scenes = annotatedScenes, actors = annotatedActors)
      }
      ```
